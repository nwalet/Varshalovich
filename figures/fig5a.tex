```latex
\documentclass[tikz,border=6pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{3d,calc,arrows.meta,quotes}

\begin{document}

\begin{tikzpicture}[
    tdplot_main_coords,
    >=Latex,
    line join=round,
    line cap=round,
    axis/.style={very thick,->},
    vector/.style={very thick,->},
    construction/.style={thick,dashed},
    every node/.style={font=\small}
]

% Coordinate system
\draw[axis] (0,0,0) -- (0,0,5.2) node[above] {$z$};
\draw[axis] (0,0,0) -- (5.6,0,0) node[right] {$y$};
\draw[axis] (0,0,0) -- (-3.8,0,0) node[below left] {$x$};

% Main vectors
\coordinate (O)  at (0,0,0);
\coordinate (r1) at (0.2,2.7,1.1);
\coordinate (r2) at (-2.0,0.8,2.7);
\coordinate (R)  at ($(r1)+(r2)$);
\coordinate (r)   at (3.2,2.2,-0.7);

\draw[vector] (O) -- (r1) node[pos=.62,right] {$r_1$};
\draw[vector] (O) -- (r2) node[pos=.58,left] {$r_2$};
\draw[vector] (O) -- (R)  node[pos=.76,left] {$R$};
\draw[vector] (O) -- (r)  node[pos=.92,right] {$r$};

% Dashed parallelogram representing the plane
\coordinate (P1) at (2.4,3.0,0.4);
\coordinate (P2) at (-1.8,1.3,3.2);
\coordinate (P3) at ($(P1)+(P2)$);

\fill[gray!8]
  (O) -- (P1) -- (P3) -- (P2) -- cycle;

\draw[construction] (O) -- (P1) -- (P3) -- (P2) -- cycle;
\draw[construction] (r1) -- (P1);
\draw[construction] (r2) -- (P2);
\draw[construction] (R)  -- (P3);

% Angular constructions near the origin
\draw[very thick,->]
  (0.48,0.05,0.05)
  .. controls (0.55,0.35,0.10) and (0.28,0.65,0.35)
  .. (0.05,0.55,0.55)
  node[pos=.65,left] {$\omega_1$};

\draw[very thick,->]
  (0.18,0.15,0.35)
  .. controls (-0.15,0.40,0.55) and (-0.55,0.42,0.95)
  .. (-0.72,0.20,1.15)
  node[pos=.65,above] {$\omega_{12}$};

\draw[very thick,->]
  (0.25,0.02,0.02)
  .. controls (0.65,-0.02,-0.05) and (0.95,0.15,-0.15)
  .. (1.15,0.25,-0.25)
  node[pos=.75,below] {$\omega$};

% Origin marker

\end{tikzpicture}
```
