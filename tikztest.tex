%\plusmark/\minusmark: the "+"/"-" node-sign markers, drawn as short thin
%lines (a cross / a dash) rather than typeset as math-mode "$+$"/"$-$"
%glyphs, so they can be sized independently of the surrounding font --
%smaller than even \tiny can manage while staying crisp. #1 is the
%half-length of each arm (default 1.2pt); e.g. \plusmark[0.8pt] for a
%smaller mark, \plusmark[1.6pt] for a larger one. Drop-in replacements for
%"$+$"/"$-$" wherever they appear as node-sign labels below.
\newcommand{\plusmark}[1][2pt]{\tikz[baseline=-0.5ex]{\draw[line width=0.3pt] (-#1,0) -- (#1,0) (0,-#1) -- (0,#1);}}
\newcommand{\minusmark}[1][2pt]{\tikz[baseline=-0.5ex]{\draw[line width=0.3pt] (-#1,0) -- (#1,0);}}

%\normedge[style]{A}{B}{pos}{offset}{label text}: draws an edge from
%coordinate A to B (in tikz style [style], default jmlinem) with a node
%placed at fractional position "pos" (0-1; 0.5 = midway, 0.75 = TikZ's
%"near end") along the path. Unlike TikZ's built-in "sloped" key, the label
%text itself is never rotated -- it stays upright -- but its *position* is
%displaced by "offset" (in the picture's local units, e.g. cm) along the
%line's normal, computed via the line's own angle (calc+atan2). A positive
%"offset" moves the label to the left of the A->B direction, negative to
%the right; flip the sign per call to control which side of the line the
%label lands on.
%The [style] slot is spliced straight into \draw[...], so it also doubles
%as the way to move the *arrowhead*, independently of the label: jmlinem
%and bjmlinem both default to a centred arrowhead (position 0.5, exactly
%as before) but now take an optional fractional position, e.g.
%\normedge[jmlinem=0.3]{...} puts the arrowhead at 0.3 instead of the
%middle; \normedge[bjmlinem=0.3]{...} likewise shifts bjmlinem's pair of
%arrowheads, keeping their usual spacing around the new centre. Both keys
%still work exactly as before when used bare (no "=value"), everywhere
%else they're used in this file.
\newcommand{\normedge}[6][jmlinem]{%
\draw[#1] let \p1=(#2), \p2=(#3), \n1={atan2(\y2-\y1,\x2-\x1)} in
  (\p1) -- (\p2) node[pos=#4,shift={(\n1+90:#5)}] {#6};
}
%\ninejdgen{s1,...,s9,SIGNS,DIRS}: fully general form of the Table 11.6
%9j hexagon. Items 1-9 are the labels by structural slot (named after
%where the standard basis diagram of {a1 a2 a3; c1 c2 c3; b1 b2 b3} puts
%its own labels):
% 1 = upper-left edge (n1-n0), "a1", 
% 2 = vertical diameter (n0-n3), "a2",
% 3 = top right edge (n1-n5), "a3"
% 4 = left side (n2-n1), "b1"
% 5 = lower left edge (n2-n3), "b2"
% 6 = oblique diameter (n2-n5), "b3"
% 7 = oblique diameter (n4-n1), "c1"
% 8 = lower right (n4-n3), "c2"
% 9 = right (n4-n5), "c3"
% 10 six sign characters for the nodes n0-n5 (standard pattern "+-+-+-"). 
% Item 11 = nine direction characters for the (see 1-9)(slot order): 
% "d" = the book's standard direction as specified above
% "r" = reversed in place. 
% The basis is \ninejdgen{a_1,a_2,a_3,b_1,b_2,b_3,c_1,c_2,c_3,+-+-+-,ddddddddd}.
%(\ninejd, kept unchanged, draws the differently-labelled Table 11.2
% hexagon of {a1 b1 c1; a2 b2 c2; a3 b3 c3}.) Needs reconciling

% sign glyph for a '+'/'-' character
\def\vmksign#1{\if+#1\plusmark\else\minusmark\fi}
% \esplit{<target cs>}{<expandable text>}: expand the text, then split it
% into a comma list with \makeclist.  Needed because \makeclist tokenises
% its argument literally, so \argx{n} has to be expanded first.
\newcommand{\esplit}[2]{%
  \edef\sl@e{#2}%
  \expandafter\makeclist\expandafter[\expandafter#1\expandafter]\expandafter{\sl@e}}
% \ifdir{<k>}{<forward>}{<reverse>}: run <forward> if direction char k is 'd'
\newcommand{\ifdir}[3]{\if\csname dir#1\endcsname d#2\else#3\fi}

\newlistcommand{\ninejdgen}{%
{\begin{tikzpicture}[baseline=0pt, inner sep=0.1em,font=\scriptsize]
  % split the sign string (arg 10) and direction string (arg 11)
  \esplit{\signlst}{\argx{10}}%
  \esplit{\direcs}{\argx{11}}%
  % six hexagon vertices n0..n5 (n0 at the top, going counter-clockwise)
  \foreach \i in {0,...,5}{\coordinate (n\i) at (90+60*\i:0.8);}%
  % a signed dot at each vertex -- count from 0 so the index matches n\i
  \foreach \s [count=\i from 0] in \signlst {%
    \node[dotnode,label={90+60*\i:\vmksign{\s}}] at (n\i) {};%
  }%
  % stash each direction character in \dir0..\dir8 for per-slot testing.
  % \global because \foreach runs its body in a group.
  \foreach \d [count=\k from 0] in \direcs {%
    \global\expandafter\let\csname dir\k\endcsname\d}%
  % edges: fixed geometry per slot; \ifdir chooses forward vs reverse
  \ifdir{0}{\normedge{n1}{n0}{0.5}{0.2}{$\argx{1}$}}{\normedge{n0}{n1}{0.5}{-0.2}{$\argx{1}$}}%
  \ifdir{1}{\normedge[jmlinem=0.25]{n0}{n3}{0.25}{0.2}{$\argx{2}$}}{\normedge[jmlinem=0.75]{n3}{n0}{0.75}{-0.2}{$\argx{2}$}}%
  \ifdir{2}{\normedge{n0}{n5}{0.5}{0.2}{$\argx{3}$}}{\normedge{n5}{n0}{0.5}{-0.2}{$\argx{3}$}}%
  \ifdir{3}{\normedge{n2}{n1}{0.5}{0.2}{$\argx{4}$}}{\normedge{n1}{n2}{0.5}{-0.2}{$\argx{4}$}}%
  \ifdir{4}{\normedge{n2}{n3}{0.5}{-0.2}{$\argx{5}$}}{\normedge{n3}{n2}{0.5}{0.2}{$\argx{5}$}}%
  \ifdir{5}{\normedge[jmlinem=0.25]{n2}{n5}{0.25}{0.2}{$\argx{6}$}}{\normedge[jmlinem=0.75]{n5}{n2}{0.75}{-0.2}{$\argx{6}$}}%
  \ifdir{6}{\normedge[jmlinem=0.25]{n4}{n1}{0.25}{0.2}{$\argx{7}$}}{\normedge[jmlinem=0.75]{n1}{n4}{0.75}{-0.2}{$\argx{7}$}}%
  \ifdir{7}{\normedge{n4}{n3}{0.5}{0.2}{$\argx{8}$}}{\normedge{n3}{n4}{0.5}{-0.2}{$\argx{8}$}}%
  \ifdir{8}{\normedge{n4}{n5}{0.5}{-0.2}{$\argx{9}$}}{\normedge{n5}{n4}{0.5}{0.2}{$\argx{9}$}}%
\end{tikzpicture}}%
}
