\chapter{MATRIX ELEMENTS OF IRREDUCIBLE TENSOR OPERATORS}
\section{THE WIGNER-ECKART THEOREM AND THE EVALUATION OF MATRIX ELEMENTS}
In this section we consider the evaluation of matrix elements of irreducible tensor operators in a general form.\\
Let a quantum mechanical system be characterized by a set of the quantum numbers ( \(n j m\) ), where \(j\) and \(m\) are the angular momentum and its projection on the quantization axis, respectively. The symbol \(n\) stands for all other quantum numbers required to define the state of the system. Let \(\Psi_{n j m}\) be the wave function of this state. The matrix element of any irreducible tensor operator \(\widehat{\mathfrak{M}}_{k}\) of rank \(k\) between the states \(\left(n^{\prime} j^{\prime} m^{\prime}\right)\) and ( \(n j m\) ) is defined as
\begin{equation}
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{ \hat{\mathfrak{M}}_{k c}}{n j m}=\int \Psi_{n^{\prime} j^{\prime} m^{\prime}}^{*} \hat{\mathfrak{M}}_{k c} \Psi_{n j m} d \tau ,\label{eq:13:1:1}
\end{equation}
where \(\widehat{\mathfrak{M}}_{k c}\) is a tensor component.

\subsection{Wigner-Eckart Theorem}
The Wigner-Eckart theorem is widely used for evaluation of matrix elements. According to this theorem, the dependence of any matrix element on the orientation of the coordinate system, i.e., on the projections \(m\), \(m^{\prime}\) and \(\kappa\), is entirely included in the \(3 j m\) symbol or the Clebsch-Gordan coefficient,
\begin{equation}
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{ \widehat{\mathfrak{M}}_{k \kappa}}{n j m}=(-1)^{j^{\prime}-m^{\prime}}\left(\begin{array}{ccc}
j^{\prime} & k & j  \\
-m^{\prime} & \kappa & m
\end{array}\right)\braOketred{ n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}=(-1)^{2 k} 
\clebsch{j}{m}{k}{\kappa}{j^{\prime}}{m^{\prime}} \frac{\braOketred{ n^{\prime} j^{\prime}}{\hat{\mathfrak{M}}_{k}}{ n j}}{\sqrt{2 j^{\prime}+1}}. \label{eq:13:1:2}
\end{equation}

The invariant factor \(\braOketred{ n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}\) is called the reduced matrix element of the set of tensor operators \(\widehat{\mathfrak{M}}_{k \kappa}\).\\
As follows from the definition \eqref{eq:13:1:2}, the reduced matrix element of the unit operator is given by
\begin{equation}
\braOketred{ n^{\prime} j^{\prime}}{\hat{\vect{I}}}{ n j}=\sqrt{2 j+1} \delta_{n^{\prime} n^{\prime}} \delta_{j^{\prime} j}.  \label{eq:13:1:3}
\end{equation}
In this section, as well as in Chap.~\ref{chap:12} we shall again use the notation
\begin{equation*}
\Fact{a b \ldots c} \equiv[(2 a+1)(2 b+1) \ldots(2 c+1)]^{\frac{1}{2}}.
\end{equation*}

\subsection{Sum Rules}
Making use of the definition \eqref{eq:13:1:2}, one can obtain the following rules for sums of matrix elements of irreducible tensor operators:
\begin{align}
\sum_{m \kappa}\left|\braOket{n^{\prime} j^{\prime}m^{\prime}}{\widehat{\mathfrak{M}}_{k \kappa}}{ n j m}\right|^{2} & =\frac{\left|\braOketred{n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}\right|^{2}}{2 j^{\prime}+1},  \label{eq:13:1:4}\\
\sum_{m^{\prime} \kappa}\left|\braOket{n^{\prime} j^{\prime}m^{\prime}}{\widehat{\mathfrak{M}}_{k \kappa}}{ n j m}\right|^{2} & =\frac{\left|\braOketred{n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}\right|^{2}}{2 j+1},  \label{eq:13:1:5}\\
\sum_{m m^{\prime}}\left|\braOket{n^{\prime} j^{\prime}m^{\prime}}{\widehat{\mathfrak{M}}_{k \kappa}}{ n j m}\right|^{2} & =\frac{\left|\braOketred{n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}\right|^{2}}{2 k+1},  \label{eq:13:1:6}\\
\sum_{m m^{\prime} \kappa}\left|\braOket{n^{\prime} j^{\prime}m^{\prime}}{\widehat{\mathfrak{M}}_{k \kappa}}{ n j m}\right|^{2} & =\left|\braOketred{n^{\prime} j^{\prime}}{\widehat{\mathfrak{M}}_{k}}{ n j}\right|^{2}. \label{eq:13:1:7}
\end{align}
\subsection{Matrix Elements of Products of Irreducible Tensor Operators}
From two irreducible tensor operators \(\widehat{P}_{a \alpha}\) and \(\widehat{Q}_{b \beta}\) one can compose the direct or reducible product \(\widehat{P}_{a \alpha}\, \widehat{Q}_{b \beta}\) and the irreducible product \(\left\{\hat{\vect{P}}_{a} \otimes \hat{\vect{Q}}_{b}\right\}_{c \gamma}\) of rank \(c\) (see Chap.~\ref{chap:3}). A matrix element of the direct tensor product may be written in the form
\begin{equation}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\hat{P}_{a \alpha} \, \hat{Q}_{b \rho}}{n j m}=\frac{(-1)^{2 a+2 b}}{\Fact{j^{\prime}}} \sum_{n_{1} j_{1} m_{1}} 
\clebsch{j_{1}}{m_{1}}{a}{\alpha}{j^{\prime}}{m^{\prime}} 
\clebsch{j}{m}{b}{\beta}{j_{1}}{m_{1}} \frac{1}{\Fact{j_1}}
\braOketred{n^{\prime} j^{\prime}}{\hat{P}_{a}}{n_{1} j_{1}}
\braOketred{n_{1} j_{1}}{\hat{Q}_{b}}{n j} . \label{eq:13:1:8}
\end{equation}
Another frequently used relationship for the same matrix element is as follows:
\begin{equation}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\hat{P}_{a \alpha} \, \hat{Q}_{b \rho}}{n j m}=\frac{(-1)^{j^{\prime}+j}}{\Fact{j^{\prime}}} \sum_{c \gamma}(-1)^{-c} \Fact{c} 
\clebsch{j}{m}{c}{\gamma}{j^{\prime}}{m^{\prime}} 
\clebsch{a}{\alpha}{b}{\beta}{c}{\gamma} 
\sum_{n_{1} j_{1}}
\sixj{a}{b}{c}{j}{j^\prime}{j_1}
\braOketred{n^\prime j^{\prime}}{\hat{P}_{a}}{n_{1} j_{1}}
\braOketred{n_{1} j_{1}}{\hat{Q}_{b}}{n j} .\label{eq:13:1:9}
\end{equation}

Matrix elements of irreducible tensor products may be evaluated from the equation
\begin{multline}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\left\{\hat{P}_{a} \otimes \hat{Q}_{b}\right\}_{c\gamma}}{n j m}\\
=(-1)^{j^{\prime}+j-c} \frac{\Fact{c}}{\Fact{j}} 
\clebsch{j}{m}{c}{\gamma}{j^{\prime}}{m^{\prime}} 
\sum_{n_{1} j_{1}}
\sixj{a}{b}{c}{j}{j^\prime}{j_1}
\braOketred{n^\prime j^{\prime}}{\hat{P}_{a}}{n_{1} j_{1}}
\braOketred{n_{1} j_{1}}{\hat{Q}_{b}}{n j} . \label{eq:13:1:10}
\end{multline}
Putting \(c=0\) and \(b=a\) in Eq. \eqref{eq:13:1:10} and multiplying the resulting expression by \((-1)^{-a} \Fact{a}\) (in accordance with Eq.~\eqref{eq:3:1:34}) we get a matrix element of scalar product of two tensors of rank a
\begin{equation}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\hat{P}_{a} \cdot \hat{Q}_{a}}{n j m}
=\delta_{j j^{\prime}} \delta_{m m^{\prime}} \frac{1}{\Fact{j}^{2}} \sum_{n_{1} j_{1}}(-1)^{-j+j_{1}}\braOketred{n^\prime j}{\hat{P}_{a}}{n_{1} j_{1}}
\braOketred{n_{1} j_{1}}{\hat{Q}_{a}}{n j} . \label{eq:13:1:11}
\end{equation}
From the three irreducible tensor operators \(\widehat{P}_{a \alpha}\), \(\widehat{Q}_{b \beta}\) and \(\widehat{R}_{d \delta}\), one can compose the products

\begin{equation*}
\begin{array}{ccc}
\hat{P}_{a a} \, \hat{Q}_{b \beta} \, \hat{R}_{d b}, & \left\{\hat{\vect{P}}_{a} \otimes \hat{Q}_{b}\right\}_{c 7} \, \hat{R}_{d b}, & \hat{P}_{a \alpha} \,\left\{\hat{Q}_{b} \otimes \hat{\vect{R}}_{d}\right\}_{f \varphi}, \\
\left\{\left\{\hat{\vect{P}}_{a} \otimes \hat{Q}_{b}\right\}_{c} \otimes \hat{\vect{R}}_{d}\right\}_{f\varphi}, & \left\{\hat{\vect{P}}_{a} \otimes\left\{\hat{Q}_{b} \otimes \hat{\vect{R}}_{d}\right\}_{c}\right\}_{f \varphi} .
\end{array}
\end{equation*}

Making permutations of operators one can construct other products which differ from the given above only by commutators of operators. The matrix element of the direct product of three tensor operators is given by
\begin{gather}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\hat{P}_{a \alpha} \, \hat{Q}_{b \beta} \, \hat{R}_{d \delta}}{n j m}=\frac{(-1)^{2 a+2 b+2 d}}{\Fact{j^{\prime}}} \sum_{n_{1} j_{1} n_{2} j_{2}} \frac{1}{\Fact{j_{1} j_{2}}}
\braOketred{n^\prime j;}{\hat{P}_{a}}{n_{1} j_{1}}
\braOketred{n_{1} j_{1}}{\hat{Q}_{b}}{n_2 j_2}\notag\\
\times
\braOketred{n_{2} j_{2}}{\hat{R}_{d}}{n j}
\sum_{m_{1} m_{2}} 
\clebsch{j_{1}}{m_{1}}{a}{\alpha}{j^{\prime}}{m^{\prime}} 
\clebsch{j_{2}}{m_{2}}{b}{\beta}{j_{1}}{m_{1}} 
\clebsch{j}{m}{d}{\delta}{j_{2}}{m_{2}}. \label{eq:13:1:12}
\end{gather}
The evaluation of matrix elements of the products \(\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c \gamma} \, \widehat{R}_{d \delta}\) and \(\widehat{P}_{a \alpha} \,\left\{\widehat{\vect{Q}}_{b} \otimes \widehat{\vect{R}}_{d}\right\}_{c \gamma}\) is reduced to the successive use of Eqs. \eqref{eq:13:1:8}-\eqref{eq:13:1:10}.

The matrix element of the irreducible tensor product of three tensors may be represented in the form
\begin{gather}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes \widehat{\vect{R}}_{d}\right\}_{f \varphi}}{n j m}
=(-1)^{f+c-j} \frac{\Fact{f c}}{\Fact{j^{\prime}}} 
\clebsch{j}{m}{f}{\varphi}{j^{\prime}}{m^{\prime}} \notag\\
\times \sum_{n_{1} j_{1} n_{2} j_{2}}(-1)^{j_{2}}\sixj{a}{b}{c}{j_{2}}{j^{\prime}}{j_{1}}\sixj{d}{c}{f}{j^{\prime}}{j}{j_{2}}\braOketred{ n^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n j} \label{eq:13:1:13}
\end{gather}
By the use of Eq.~\eqref{eq:3:3:1} the evaluation of the matrix elements of \(\left\{\widehat{\vect{P}}_{a} \otimes\left\{\widehat{\vect{Q}}_{b} \otimes \widehat{\vect{R}}_{d}\right\}_{c}\right\}_{f \varphi}\) is reduced to applying Eq. \eqref{eq:13:1:13}.

The matrix element of a scalar product may be obtained from Eq. \eqref{eq:13:1:13}. Taking into account the factor \((-1)^{-c} \Fact{c}\), one gets
\begin{gather}
\braOket{n^{\prime} j^{\prime} m^{\prime}}{\left(\left\{\hat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \cdot \widehat{\vect{R}}_{c}\right)}{n j m}=\delta_{j j^{\prime}} \delta_{m m^{\prime}}(-1)^{2 j-c} \frac{\Fact{c}}{\Fact{j}^{2}} \notag\\
\times \sum_{n_{1} j_{1} n_{2} j_{2}}
\sixj{a}{b}{c}{j_{2}}{j}{j_{1}}
\braOketred{ n^{\prime} j}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}
\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}
\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{c}}{ n j} . \label{eq:13:1:14}
\end{gather}
Of the four irreducible tensors \(\widehat{P}_{a \alpha}, \widehat{Q}_{b \beta}, \widehat{R}_{d \delta}\) and \(\widehat{S}_{e s}\) one can compose the following types of tensor products:
\begin{enumerate}[label=(\alph*)]
\item \(\widehat{P}_{a \alpha} \, \widehat{Q}_{b \beta} \, \widehat{R}_{d \delta} \, \widehat{S}_{c \varepsilon}\);
\item \(\left\{\hat{\vect{P}}_{a} \otimes \hat{\vect{Q}}_{b}\right\}_{c \gamma} \, \widehat{R}_{d \delta} \, \widehat{S}_{e c}\), \(\widehat{P}_{a \alpha} \,\left\{\widehat{\vect{Q}}_{b} \otimes \widehat{\vect{R}}_{d}\right\}_{k c} \, \widehat{S}_{e c}\), \(\widehat{P}_{a \alpha} \, \widehat{Q}_{b \beta} \,\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{h \sigma}\);
\item \(\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes \widehat{\vect{R}}_{d}\right\}_{f \varphi} \, \widehat{S}_{e c}\), \(\left\{\widehat{\vect{P}}_{a} \otimes\left\{\widehat{\vect{Q}}_{b} \otimes \widehat{\vect{R}}_{d}\right\}_{h}\right\}_{f \varphi} \, \widehat{S}_{e c}\),  \(\widehat{P}_{a \alpha} \,\left\{\left\{\widehat{\vect{Q}}_{b} \otimes \widehat{\vect{R}}_{d}\right\}_{h} \otimes \widehat{\vect{S}}_{e}\right\}_{f \varphi}\), \par \(\widehat{P}_{a \alpha} \,\left\{\widehat{\vect{Q}}_{b} \otimes\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{h}\right\}_{f \varphi}\), \(\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c q} \,\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{h \sigma}\);
\item \(\left\{\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes \widehat{\vect{R}}_{d}\right\}_{f} \otimes \widehat{\vect{S}}_{e}\right\}_{k c}\), \(\left\{\widehat{\vect{P}}_{a} \otimes\left\{\widehat{\vect{Q}}_{b} \otimes\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{h}\right\}_{f}\right\}_{k}\), \(\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{f}\right\}_{k c}\).
\end{enumerate}
Products with other orders of the operators can be reduced to the products listed here using the commutation rules.

A matrix element of a direct product may be evaluated as follows:
\begin{gather}
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{ \widehat{P}_{a \alpha} \widehat{Q}_{b \beta} \widehat{R}_{d \delta} \widehat{S}_{e \varepsilon}}{n j m}=(-1)^{2 a+2 b+2 d+2 e} \frac{1}{\Fact{j^{\prime}}} \sum_{\substack{n_{1} n_{2} n_{3}\notag \\
j_{1} j_{2} j_{3}}}\braOketred{ n^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n_{3} j_{3}} \\
\times\braOketred{ n_{3} j_{3}}{\widehat{\vect{S}}_{e}}{ n j} \frac{1}{\Fact{j_{1} j_{2} j_{3}}} \sum_{m_{1} m_{2} m_{3}} 
\clebsch{j_{1}}{m_{1}}{a}{\alpha}{j^{\prime}}{m^{\prime}} 
\clebsch{j_{2}}{m_{2}}{b}{\beta}{j_{1}}{m_{1}} 
\clebsch{j_{2}}{m_{3}}{d}{\delta}{j_{2}}{m_{2}} 
\clebsch{j_{3}}{m}{e}{\varepsilon}{j_{3}}{m_{3}}. \label{eq:13:1:15}
\end{gather}
Matrix elements of products of (b)- and (c)-type products can be evaluated by using Eqs. \eqref{eq:13:1:8}-\eqref{eq:13:1:14}.

The evaluation of matrix elements of irreducible tensor products yields
\begin{gather}
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{\left\{\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes \widehat{\vect{R}}_{d}\right\}_{f} \otimes \widehat{\vect{S}}_{e}\right\}_{k c}}{n j m}=(-1)^{j-j^{\prime}+f+c-k} \frac{\Fact{f c k}}{\Fact{j^{\prime}}} 
\clebsch{j}{m}{k}{c}{j^{\prime}}{m^{\prime}} \sum_{\substack{n_{1} n_{2} n_{3} \\
j_{1} j_{2} j_{3}}}(-1)^{j_{2}+j_{3}}\sixj{a}{b}{c}{j_{2}}{j^{\prime}}{j_{1}} \notag\\
\times\sixj{d}{c}{f}{j^{\prime}}{j_{3}}{j_{2}}\sixj{e}{f}{k}{j^{\prime}}{j}{j_{3}}\braOketred{ n^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n_{3} j_{3}}\braOketred{ n_{3} j_{3}}{\widehat{\vect{S}}_{e}}{ n j},  \label{eq:13:1:16}\\
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{\left(\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes \widehat{\vect{R}}_{d}\right\}_{f} \cdot \widehat{\vect{S}}_{f}\right)}{n j m}=(-1)^{c-f-j} \frac{\Fact{f c}}{\Fact{j}^{2}} \delta_{j^{\prime} j} \delta_{m^{\prime} m} \sum_{\substack{n_{1} n_{2} n_{3} \\
j_{1} j_{2} j_{3}}}(-1)^{j_{2}} \cdot\sixj{a}{b}{c}{j_{2}}{j}{j_{1}}\sixj{d}{c}{f}{j}{j_{3}}{j_{2}} \notag\\
\times\braOketred{ n^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n_{3} j_{3}}\braOketred{ n_{3} j_{3}}{\widehat{\vect{S}}_{f}}{ n j}. \label{eq:13:1:17}
\end{gather}
Finally Eq.~\eqref{eq:3:3:5} allows one to simplify the calculation of matrix elements of the tensor product \(\left\{\widehat{\vect{P}}_{a} \otimes\left\{\widehat{\vect{Q}}_{b} \otimes\right.\right.\)\(\left.\left.\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{h}\right\}_{f}\right\}_{k \kappa}\) to Eq. \eqref{eq:13:1:16}. Thus
\begin{gather}
\braOket{ n^{\prime} j^{\prime} m^{\prime}}{\left\{\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \otimes\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{j}\right\}_{k c}}{n j m}=(-1)^{2 j^{\prime}+c-k-f} \frac{\Fact{c f k}}{\Fact{j^{\prime}}} 
\clebsch{j^{\prime}}{m}{k}{c}{j^{\prime}}{m^{\prime}} \sum_{\substack{n_{1} n_{2} n_{3} \notag\\
j_{1} j_{2} j_{3}}}\sixj{a}{b}{c}{j_{2}}{j^{\prime}}{j_{1}}\sixj{d}{e}{f}{j}{j_{2}}{j_{3}} \\
\times\sixj{f}{c}{k}{j^{\prime}}{j}{j_{2}}\braOketred{ n^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n_{3} j_{3}}\braOketred{ n_{3} j_{3}}{\widehat{\vect{S}}_{e}}{ n j},  \label{eq:13:1:18}
\end{gather}
\begin{gather}
%\left.
    \braOket{n^{\prime} j^{\prime} m^{\prime}}{\left(\left\{\widehat{\vect{P}}_{a} \otimes \widehat{\vect{Q}}_{b}\right\}_{c} \cdot\left\{\widehat{\vect{R}}_{d} \otimes \widehat{\vect{S}}_{e}\right\}_{c}\right)}{n j m}=\delta_{j^{\prime}} \delta_{m^{\prime} m} \frac{\Fact{c}^{2}}{\Fact{j}^{2}} \sum_{\substack{n_{1} n_{2} n_{3}notag \\
j_{1} j_{2} j_{3}}}(-1)^{-j+j_{2}}\sixj{a}{b}{c}{j_{2}}{j}{j_{1}}\sixj{d}{e}{c}{j}{j_{2}}{j_{3}} \notag\\
\times
\braOketred{ n^{\prime} j}{\widehat{\vect{P}}_{a}}{ n_{1} j_{1}}
\braOketred{ n_{1} j_{1}}{\widehat{\vect{Q}}_{b}}{ n_{2} j_{2}}
\braOketred{ n_{2} j_{2}}{\widehat{\vect{R}}_{d}}{ n_{3} j_{3}}
\braOketred{ n_{3} j_{3}}{\widehat{\vect{S}}_{e}}{ n j} . \label{eq:13:1:19}
\end{gather}
\subsection{Matrix Elements of Operators Which Depend on Variables of Two Subsystems}
If a quantum-mechanical system consists of two subsystems, 1 and 2 , and eigenstates of these subsystems are characterized by the sets of quantum numbers \(\left(n_{1} j_{1} m_{1}\right)\) and \(\left(n_{2} j_{2} m_{2}\right)\), it is often convenient to introduce the two following representations of matrix elements.\\
(a) The angular momenta of both subsystems are not coupled. Then
\begin{equation}
\braOket{n_{1}^{\prime} j_{1}^{\prime} m_{1}^{\prime} ; n_{2}^{\prime} j_{2}^{\prime} m_{2}^{\prime}}{\widehat{\vect{P}}(1,2)}{n_{1} j_{1} m_{1} ; n_{2} j_{2} m_{2}}
=\int \Psi_{n_{1}^{\prime} j_{1}^{\prime} m_{1}^{\prime}}^{*}(1) \Psi_{n_{2}^{\prime} j_{2}^{\prime} m_{2}^{\prime}}^{*}(2) \widehat{\vect{P}}(1,2) \Psi_{n_{1} j_{1} m_{1}}(1) \Psi_{n_{2} j_{2} m_{2}}(2) d r_{1} d \tau_{2} \label{eq:13:1:20}
\end{equation}
where \(\Psi_{n_{1} j_{1} m_{1}}\) and \(\Psi_{n_{2} j_{2} m_{2}}\) are wave functions that describe the eigenstates of each subsystem. Such a representation of matrix elements is called the \(\left(n_{1} j_{1} m_{1} ; n_{2} j_{2} m_{2}\right)\)-representation.\\
(b) If angular momenta \(\mathrm{j}_{1}\) and \(\mathrm{j}_{2}\) are coupled to the resultant angular momentum j with projection \(m= m_{1}+m_{2}\), then
\begin{equation}
\braOket{ n_{1}^{\prime} j_{1}^{\prime} n_{2}^{\prime} j_{2}^{\prime} j^{\prime} m^{\prime}}{\widehat{\vect{P}}(1,2)}{n_{1} j_{1} n_{2} j_{2} j m}=\int \Psi_{n_{1}^{\prime} j_{1}^{\prime} n_{2}^{\prime} j_{2}^{\prime} j^{\prime} m^{\prime}}^{*}(1,2) \widehat{\vect{P}}(1,2) \Psi_{n_{1} j_{1} n_{2} j_{2} j m}(1,2) d \tau_{1} d \tau_{2} \label{eq:13:1:21}
\end{equation}
Here
\begin{equation}
\Psi_{n_{1} j_{1} n_{2} j_{2} j m}(1,2)=\sum_{m_{1} m_{2}} 
\clebsch{j_{1}}{m_{1}}{j_{2}}{m_{2}}{j}{m} \Psi_{n_{1} j_{1} m_{1}}(1) \Psi_{n_{2} j_{2} m_{2}}(2) \label{eq:13:1:22}
\end{equation}
This representation of matrix elements is called the \(\left(n_{1} j_{1} n_{2} j_{2} j m\right)\)-representation.

In general case, we have
\begin{gather}
\braOket{ n_{1}^{\prime} j_{1}^{\prime} m_{1}^{\prime} ; n_{2}^{\prime} j_{2}^{\prime} m_{2}^{\prime}}{ \widehat{\vect{P}}_{a \alpha}(1,2)}{n_{1} j_{1} m_{1} ; n_{2} j_{2} m_{2}} \notag\\
=(-1)^{2 a} \frac{1}{\Fact{j^{\prime}}} \sum_{j m j^{\prime} m^{\prime}} 
\clebsch{j_{1}}{m_{1}}{j_{2}}{m_{2}}{j}{m} 
\clebsch{j_{1}^{\prime}}{m_{1}^{\prime}}{j_{2}^{\prime}}{m_{2}^{\prime}}{j^{\prime}}{m^{\prime}} 
\clebsch{j}{m}{a}{\alpha}{j^{\prime}}{m^{\prime}}\braOketred{ n_{1}^{\prime} j_{1}^{\prime} n_{2}^{\prime} j_{2}^{\prime} j^{\prime}}{\widehat{\vect{P}}_{a}(1,2)}{ n_{1} j_{1} n_{2} j_{2} j} \label{eq:13:1:23}
\end{gather}
Any operator \(\widehat{A}(1,2)\) which depends on variables of two subsystems can frequently be expressed as a direct or irreducible product of two operators \(\widehat{P}_{a \alpha}(1)\) and \(\widehat{Q}_{b \beta}(2)\) which depend only on variables of the first and second subsystem, respectively.

\begin{equation*}
\widehat{A}(1,2)=\widehat{P}_{a \alpha}(1) \widehat{Q}_{b \beta}(2) \quad \text { or } \quad \widehat{A}(1,2)=\left\{\widehat{\vect{P}}_{a}(1) \otimes \widehat{\vect{Q}}_{b}(2)\right\}_{c \gamma}
\end{equation*}

The matrix element for the direct-product case is given by
\begin{gather}

\end{gather}

\end{gather}

\begin{gather}

\subsection{Matrix Elements of Operators Which Depend on 
 Variables of One of the Subsystems}
Let \(\widehat{P}_{a \alpha}(1)\) be an operator which depends on variables of subsystem 1. Matrix elements of this operator are diagonal with respect to quantum numbers which characterize eigenstates of subsystem 2 .
\begin{equation}
\braOket{ n_{1}^{\prime} j_{1}^{\prime} m_{1}^{\prime} ; n_{2}^{\prime} j_{2}^{\prime} m_{2}^{\prime}}{ \widehat{P}_{a \alpha}(1)}{n_{1} j_{1} m_{1} ; n_{2} j_{2} m_{2}}=\delta_{n_{2}^{\prime} n_{2}} \delta_{j_{2}^{\prime} j_{2}} \delta_{m_{2}^{\prime} m_{2}}\braOket{ n_{1}^{\prime} j_{1}^{\prime} m_{1}^{\prime}}{ \widehat{P}_{a \alpha}(1)}{n_{1} j_{1} m_{1}} \label{eq:13:1:39}

\times \frac{\Fact{c}}{\Fact{j_{1}}^{2}} \sum_{N_{1} N_{2} J}\sixj{a}{b}{c}{J_{2}}{j_{1}}{J_{1}}\braOketred{ n_{1}^{\prime} j_{1}}{\widehat{\vect{P}}_{a}(1)}{ N_{1} J_{1}}\braOketred{ N_{1} J_{1}}{\widehat{\vect{Q}}_{b}(1)}{ N_{2} J_{2}}\braOketred{ N_{2} J_{2}}{\widehat{\vect{R}}_{c}(1)}{ n_{1} j_{1}} \label{eq:13:1:45}
\end{gather}

\times\braOketred{ n_{1}^{\prime} j_{1}^{\prime}}{\widehat{\vect{P}}_{a}(1)}{ N_{1} J_{1}}\braOketred{ N_{1} J_{1}}{\widehat{\vect{Q}}_{b}(1)}{ N_{2} J_{2}}\braOketred{ N_{2} J_{2}}{\widehat{\vect{R}}_{d}(1)}{ N_{3} J_{3}}\braOketred{N_{3} J_{3}}{\widehat{\vect{S}}_{e}(1)}{ n_{1} j_{1}}, \label{eq:13:1:48}
\end{gather}

\braOket{ n_{1}^{\prime} j_{1}^{\prime} n_{2}^{\prime} j_{2}^{\prime} j^{\prime} m^{\prime}}{\left(\left\{\left\{\widehat{\vect{P}}_{a}(1) \otimes \widehat{\vect{Q}}_{b}(1)\right\}_{c} \otimes \widehat{\vect{R}}_{d}(1)\right\}_{f} \cdot \widehat{\vect{S}}_{f}(1)\right)}{n_{1} j_{1} n_{2} j_{2} j m} \notag\\
=\delta_{n_{2}^{\prime} n_{2}} \delta_{j_{1}^{\prime} j_{1}} \delta_{j_{2}^{\prime} j_{2}} \delta_{j^{\prime} j^{\prime}} \delta_{m^{\prime} m}(-1)^{c-f-j_{1}} \frac{\Fact{f c}}{\Fact{j_{1}}^{2}} \sum_{\substack{N_{1} N_{2} N_{3} \\
J_{1} J_{2} J_{3}}}(-1)^{J_{2}}\sixj{a}{b}{c}{J_{2}}{j_{1}}{J_{1}}\sixj{d}{c}{f}{j_{1}}{J_{3}}{J_{2}} \notag\\
\times\braOketred{ n_{1}^{\prime} j_{1}}{\widehat{\vect{P}}_{a}(1)}{ N_{1} J_{1}}\braOketred{ N_{1} J_{1}}{\widehat{\vect{Q}}_{b}(1)}{ N_{2} J_{2}}\braOketred{ N_{2} J_{2}}{\widehat{\vect{R}}_{d}(1)}{ N_{3} J_{3}}\braOketred{ N_{3} J_{3}}{\widehat{\vect{S}}_{f}(1)}{ n_{1} j_{1}} \label{eq:13:1:49}
\end{gather}
Matrix elements for tensor products with other coupling schemes of operators may be obtained from the foregoing equations by the use of the recoupling rules and the commutation relations for irreducible tensors (Chap.~\ref{chap:3}).

\section{MATRIX ELEMENTS OF BASIC TENSOR OPERATORS}
\subsection{Some Introductory Remarks}
In this section we shall consider matrix elements of the following operators and their tensor products:\\
(a) the unit operator \(\widehat{I}\);\\
(b) the unit vector operator \(\widehat{\vect{n}} \equiv \widehat{\vect{n}}_{1}\);\\
(c) the gradient operators \(\hat{\nabla} \equiv \hat{\nabla}_{1}\) and \(\hat{\nabla}_{\Omega} \equiv\left(\hat{\nabla}_{\Omega}\right)_{1}\);\\
(d) the total angular momentum operator \(\widehat{\vect{J}} \equiv \widehat{\vect{J}}_{1}\);\\
(e) the orbital angular momentum operator \(\hat{\vect{L}} \equiv \hat{\vect{L}}_{1}\);\\
(f) the spin angular momentum operator \(\widehat{\vect{S}} \equiv \widehat{\vect{S}}_{1}\);\\
(g) the spherical harmonic operator \(Y_{l m}(\vartheta, \varphi)\).

All matrix elements may be evaluated from the equations of Sec.~\ref{sec:13.1}.\\
A wave function which describes an eigenstate of some quantum system will generally depend on the position \((r, \vartheta, \varphi)\) as well as on the spin variables ( \(\xi\) ). It may be defined by
\begin{equation}
\Phi(r, \vartheta, \varphi ; \xi)=\Psi_{n}(r) Y_{l m}(\vartheta \varphi)_{s m .}(\xi)=
\braket{r, \vartheta, \varphi ; \xi}{n, l, m ; s m_{s}}. \label{eq:13:2:1}
\end{equation}
where \(l\) and \(m\) are the orbital momentum and its projection; \(s\) and \(m_{s}\) are the spin momentum and its projection. The symbol \(n\) has been defined previously in Sec.~\ref{sec:13.1.1}.

The wave functions \eqref{eq:13:2:1} determine the (nlmsms)-representation of operators.\\
If the angular momenta \(\widehat{\vect{L}}\) and \(\widehat{\vect{S}}\) are coupled into the total angular momentum \(\widehat{\vect{J}}\) with its projection \(M\), the wave function which describes an eigenstate of the quantum system is defined as
\begin{equation}
\Phi(r, \vartheta, \varphi ; \xi)=\Psi_{n}(r) \sum_{m m,} 
\clebsch{l}{m}{s}{m,}{J}{M} Y_{l m}(\vartheta, \varphi) \chi_{s m,}(\xi)=\Psi_{n}(r) Y_{J M}^{l s}(\vartheta, \varphi)=\braket{r, \vartheta, \varphi ; \xi}{ n, l, s, J, M}. \label{eq:13:2:2}
\end{equation}
These wave functions determine the ( \(n l s J M\) )-representation of operators.\\
If an operator acts only on position variables, the evaluation of matrix elements in the ( \(n\) lmsm. \(_{s}\) )-representation is reduced to calculating matrix elements in the (nlm)-representation.
\begin{equation}
\braOket{ n^{\prime} l^{\prime} m^{\prime} ; s^{\prime} m_{s}^{\prime}}{ \widehat{\mathfrak{M}}_{k \kappa}(r, \vartheta, \varphi)}{n l m ; s m_{s}}=\delta_{s \theta^{\prime}} \delta_{m_{s} m_{!}^{\prime}}\braOket{ n^{\prime} l^{\prime} m^{\prime}}{ \widehat{\mathfrak{M}}_{k \kappa}(r, \vartheta ; \varphi)}{n l m} . \label{eq:13:2:3}
\end{equation}
If an operator depends only on spin variables, we get a similar relationship
\begin{equation}
\braOket{ n^{\prime} l^{\prime} m^{\prime} ; s^{\prime} m_{s}^{\prime}}{ \hat{\mathfrak{N}}_{k \kappa}(\xi)}{n l m ; s m_{s}}=\delta_{l l} \delta_{m m^{\prime}}\braOket{ n^{\prime} s^{\prime} m_{s}^{\prime}}{ \hat{\mathfrak{N}}_{k \kappa}(\xi)}{n s m_{s}} \label{eq:13:2:4}
\end{equation}
Hereafter matrix elements of operators which depend only on position variables or only on spin variables will be evaluated in the ( \(n l m)\) - or the ( \(n s m_{s}\) )-representation, respectively.

To determine matrix elements of operators \(\widehat{\mathfrak{M}}_{k c}(r, \vartheta, \varphi)\) and \(\widehat{\mathfrak{N}}_{k c}(\xi)\) in the (nlsJM)-representation one should use the equations
\begin{align}
& \braOketred{ n^{\prime} l^{\prime} s^{\prime} J^{\prime}}{\hat{\mathfrak{M}}_{k}}{ n l s J}=\delta_{0^{\prime}}(-1)^{J+l^{\prime}+\infty+k} \Fact{J J^{\prime}}\sixj{l}{s}{J}{J^{\prime}}{k}{l^{\prime}}\braOketred{ n^{\prime} l^{\prime}}{\mathfrak{M}_{k}}{ n l}  \label{eq:13:2:5}\\
& \braOketred{ n^{\prime} l^{\prime} s^{\prime} J^{\prime}}{\hat{\mathfrak{N}}_{k}}{ n l s J}=\delta_{l l^{\prime}}(-1)^{J^{\prime}+l+a+k} \Fact{J J^{\prime}}\sixj{s}{l}{J}{J^{\prime}}{k}{s^{\prime}}\left(n^{\prime} s^{\prime}\left\|\hat{\mathfrak{N}}_{k}\right\| n s\right) \label{eq:13:2:6}
\end{align}
If some operator \(\widehat{R}_{a \alpha}(r, \vartheta, \varphi ; \xi)\) acts on position as well as on spin variables, its matrix elements will be given in the (nlsJM)-representation. These matrix elements may be evaluated in the (nlmsm,)-representation with aid of
\begin{equation}
\braOket{ n^{\prime} l^{\prime} m^{\prime} ; s^{\prime} m_{o}^{\prime}}{ \widehat{R}_{a \alpha}(r, \vartheta, \varphi ; \xi)}{n l m ; s m_{s}}=\sum_{\substack{J M \\ J^{\prime} M^{\prime}}} 
\clebsch{l}{m}{s}{m_{e}}{J}{M} 
\clebsch{l^{\prime}}{m^{\prime}}{s^{\prime}}{m_{\prime}^{\prime}}{J^{\prime}}{M^{\prime}}\braOket{n^{\prime} l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{\widehat{R}_{a \alpha}(r, \vartheta, \varphi ; \xi)}{n l s J M} \label{eq:13:2:7}
\end{equation}
or from the corresponding equations of the section above.\\
From now on, we will omit the arguments of operators in the expressions for matrix elements to simplify the equations. In addition, the index \(n\) will also be dropped in all the cases for which it is not essential.

\subsection{Matrix Elements of the Unit Operator \(\hat{\mathrm{I}}\)}
The unit operator \(\widehat{\vect{I}}\) is an irreducible tensor operator of rank zero. It does not act on variables \(r, \vartheta, \varphi, \xi\). Hence, in any representation its diagonal matrix elements are determined by
\begin{equation}
\braOket{\lambda^{\prime}}{ \hat{\vect{I}}}{\lambda}=
\braket{\lambda^{\prime}}{\lambda}=\delta_{\lambda \lambda^{\prime}} \label{eq:13:2:8}
\end{equation}
All non-diagonal matrix elements vanish. A reduced matrix element of the operator \(\hat{\vect{I}}\) may be represented in the form
\begin{equation}
\braOketred{l}{\hat{\vect{I}}}{l}=\sqrt{2 l+1}, \quad(s\|\hat{\vect{I}}\| s)=\sqrt{2 s+1}, \quad\braOketred{ l^{\prime} s^{\prime} J}{\hat{\vect{I}}}{ l s J}=\sqrt{2 J+1} \delta_{l l^{\prime}} \delta_{s a^{\prime}} \label{eq:13:2:9}
\end{equation}
\subsection{Matrix Elements of the Unit Vector \(\hat{\vect{n}}(\vartheta, \varphi) \equiv \widehat{\vect{n}}_{1}(\vartheta, \varphi)\)}
The unit vector operator \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) acts only on variables \(\vartheta, \varphi\). In the (lm)-representation matrix elements of spherical components of this operator are given by
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{ \widehat{n}_{1 \mu}}{l m}=
\frac{\braOketred{ l^{\prime}}{\widehat{n}_{1}}{ l}}{\sqrt{2 l^{\prime}+1}} 
\clebsch{l}{m}{1}{\mu}{l^{\prime}}{m^{\prime}} \label{eq:13:2:10}
\end{equation}
where
\begin{equation}
\braOketred{ l^{\prime}}{\hat{\vect{n}}_{1}}{ l}=\sqrt{2 l+1} 
\clebsch{1}{0}{1}{0}{l^{\prime}}{0} \label{eq:13:2:11}
\end{equation}
Taking into account Eq. \eqref{eq:13:2:5}, we obtain a reduced matrix element in the (lsJ)-representation

\begin{equation}
\braOketred{ l^{\prime} s^{\prime} J^{\prime}}{\hat{\vect{n}}_{1}}{ l s J}=\delta_{s e^{\prime}}(-1)^{s+J+l^{\prime}+1} \sqrt{(2 J+1)\left(2 J^{\prime}+1\right)(2 l+1)} 
\clebsch{1}{0}{1}{0}{\prime}{0}\sixj{l}{s}{J}{J^{\prime}}{1}{l^{\prime}}\label{eq:13:2:12}
\end{equation}

The matrix element \eqref{eq:13:2:10} vanishes unless \(l^{\prime}=l \pm 1\). Using Eq. \eqref{eq:13:2:11} one can rewrite Eq. \eqref{eq:13:2:10} in component form as
\begin{align}
\braOket{ l+1 m^{\prime}}{ \hat{n}_{1 \pm 1}}{l m} & =\sqrt{\frac{(l \pm m+1)(l \pm m+2)}{2(2 l+1)(2 l+3)}} \delta_{m^{\prime} m \pm 1} \notag\\
\braOket{ l+1 m^{\prime}}{ \hat{n}_{10}}{l m} & =\sqrt{\frac{(l-m+1)(l+m+1)}{(2 l+1)(2 l+3)}} \delta_{m^{\prime} m}  \label{eq:13:2:13}\\
\braOket{ l-1 m^{\prime}}{ \hat{n}_{1 \pm 1}}{l m} & =-\sqrt{\frac{(l \mp m-1)(l \mp m)}{2(2 l+1)(2 l-1)}} \delta_{m^{\prime} m \pm 1} \\
\braOket{ l-1 m^{\prime}}{ \hat{n}_{10}}{l m} & =\sqrt{\frac{(l-m)(l+m)}{(2 l+1)(2 l-1)}} \delta_{m^{\prime} m}. \notag
\end{align}
All other matrix elements of the spherical components of the operator in question are equal to zero.\\
The matrix elements of the cartesian components of the operator \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) are given by
\begin{align}
\braOket{ l^{\prime} m \pm 1}{ \widehat{n}_{x}}{l m} & =\mp \frac{1}{2} \sqrt{\frac{(l \pm m+1)(l \pm m+2)}{(2 l+1)(2 l+3)}} \delta_{l^{\prime} l+1} \pm \frac{1}{2} \sqrt{\frac{(l \mp m-1)(l \mp m)}{(2 l+1)(2 l-1)}} \delta_{l^{\prime} l-1}  \label{eq:13:2:14}\\
\braOket{ l^{\prime} m \pm 1}{ \widehat{n}_{y}}{l m} & =\frac{i}{2} \sqrt{\frac{(l \pm m+1)(l \pm m+2)}{(2 l+1)(2 l+3)}} \delta_{l^{\prime} l+1}-\frac{i}{2} \sqrt{\frac{(l \mp m-1)(l \mp m)}{(2 l+1)(2 l-1)}} \delta_{l^{\prime} l-1}  \label{eq:13:2:15}\\
\braOket{ l^{\prime} m}{ \widehat{n}_{z}}{l m} & =\sqrt{\frac{(l-m+1)(l+m+1)}{(2 l+1)(2 l+3)}} \delta_{l^{\prime} l+1}+\sqrt{\frac{(l-m)(l+m)}{(2 l+1)(2 l-1)}} \delta_{l^{\prime} l-1} \label{eq:13:2:16}
\end{align}
All other matrix elements of the cartesian components of \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) vanish.\\
It is easy to obtain the matrix elements of direct products of two operators involving \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) :
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{ \widehat{n}_{1 \mu} \widehat{n}_{1 \nu}}{l m}=\frac{(-1)^{\mu}}{3} \delta_{l l^{\prime}} \delta_{m m^{\prime}} \delta_{\mu-\nu}+\sqrt{\frac{2}{3} \frac{2 l+1}{2 l^{\prime}+1}} 
\clebsch{l}{0}{2}{0}{l^{\prime}}{0} 
\clebsch{1}{\mu}{1}{\nu}{2}{\kappa} 
\clebsch{l}{m}{2}{\kappa}{l^{\prime}}{m^{\prime}} \label{eq:13:2:17}
\end{equation}
These matrix elements vanish unless \(l^{\prime}=l, l \pm 2\). To derive the matrix elements of direct products of cartesian components of \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) one may express these products in terms of the products of spherical components. In this way we get
\begin{align}
\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{x} \hat{n}_{x}}{l m} & =\frac{1}{2}\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{1 \pm 1} \hat{n}_{1 \pm 1}}{l m} \notag\\
\braOket{ l^{\prime} m}{ \hat{n}_{x} \hat{n}_{x}}{l m} & =-\braOket{ l^{\prime} m}{ \hat{n}_{1 \pm 1} \hat{n}_{1 \mp 1}}{l m} \notag\\
\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{y} \hat{n}_{y}}{l m} & =-\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{1 \pm 1} \hat{n}_{1 \pm 1}}{l m} \notag\\
\braOket{ l^{\prime} m}{ \hat{n}_{y} \hat{n}_{y}}{l m} & =-\braOket{ l^{\prime} m}{ \hat{n}_{1 \pm 1} \hat{n}_{1 \mp 1}}{l m}\notag \\
\braOket{ l^{\prime} m}{ \hat{n}_{x} \hat{n}_{z}}{l m} & =\braOket{ l^{\prime} m}{ \hat{n}_{10} \hat{n}_{10}}{l m}  \label{eq:13:2:18}\\
\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{x} \hat{n}_{y}}{l m} & =\mp \frac{i}{2}\braOket{ l^{\prime} m \pm 2}{ \hat{n}_{l \pm 1} \hat{n}_{1 \pm 1}}{l m} \notag\\
\braOket{ l^{\prime} m \pm 1}{ \hat{n}_{x} \hat{n}_{z}}{l m} & =\mp \frac{1}{\sqrt{2}}\braOket{ l^{\prime} m \pm 1}{ \hat{n}_{1 \pm 1} \hat{n}_{10}}{l m} \notag\\
\braOket{ l^{\prime} m \pm 1}{ \hat{n}_{y} \hat{n}_{z}}{l m} & =\frac{i}{\sqrt{2}}\braOket{ l^{\prime} m \pm 1}{ \hat{n}_{1 \pm 1} \hat{n}_{10}}{l m}. \notag
\end{align}
The matrix elements of the irreducible tensor product of \(k\) unit vector operators \(\widehat{\vect{n}}_{1}(\vartheta, \varphi)\) may be represented in the form
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left\{\ldots\left\{\left\{\widehat{\vect{n}}_{1} \otimes \widehat{\vect{n}}_{1}\right\}_{q_{2}} \otimes \widehat{\vect{n}}_{1}\right\}_{q_{3}} \otimes \ldots \otimes \widehat{\vect{n}}_{1}\right\}_{q_{k} \kappa_{k}}}{l m}=\sqrt{\frac{2 l+1}{2 l^{\prime}+1}} 
\clebsch{l}{o}{q_{k}}{0}{l^{\prime}}{0} \Fact{i=2}^{k} 
\clebsch{q_{i-1}}{0}{1}{0}{q_{i}}{0} 
\clebsch{l}{m}{q_{k}}{\kappa_{k}}{l^{\prime}}{m^{\prime}} \label{eq:13:2:19}
\end{equation}
where \(q_{1}=1\). In particular, if \(q_{2}=2, q_{3}=3, q_{4}=4, \ldots\) one has
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left\{\ldots\left\{\left\{\hat{\vect{n}}_{1} \otimes \widehat{\vect{n}}_{1}\right\}_{2} \otimes \widehat{\vect{n}}_{1}\right\}_{3} \otimes \ldots \otimes \widehat{\vect{n}}_{1}\right\}_{k \kappa}}{l m}=\sqrt{\frac{k!}{(2 k-1)!!} \frac{2 l+1}{2 l^{\prime}+1}} 
\clebsch{l}{0}{k}{0}{l^{\prime}}{0} 
\clebsch{l}{m}{k}{\kappa}{l^{\prime}}{m^{\prime}} \label{eq:13:2:20}
\end{equation}
From Eq. \eqref{eq:13:2:19} one obtains two obvious equalities,
\begin{equation}
\begin{gathered}
\braOket{ l^{\prime} m^{\prime}}{(\hat{\vect{n}} \cdot \hat{\vect{n}})}{l m}=\delta_{l^{\prime}} \delta_{m^{\prime} m^{\prime}} \\
\braOket{ l^{\prime} m^{\prime}}{[\hat{\vect{n}} \times \hat{\vect{n}}]}{l m}=0
\end{gathered}
\end{equation}

\subsection{Matrix Elements of the Operator \(\hat{\nabla}(r, \vartheta, \varphi) \equiv \hat{\nabla}_{1}(r, \vartheta, \varphi)\)}
The operator \(\hat{\nabla}_{1}(r, \vartheta, \varphi)\) acts only on variables \(r, \vartheta, \varphi\). The matrix elements of its spherical components are determined by
\begin{equation}
\braOket{ n^{\prime} l^{\prime} m^{\prime}}{ \hat{\nabla}_{1 \mu}}{n l m}=
\frac{\braOketred{ n^{\prime} l^{\prime}}{\hat{\nabla}_{1}}{ n l}}{\sqrt{2 l^{\prime}+1}} 
\clebsch{l}{m}{1}{\mu}{l^{\prime}}{m^{\prime}} \label{eq:13:2:21}
\end{equation}
where
\begin{equation}
\braOketred{ n^{\prime} l^{\prime}}{\widehat{\nabla}_{1}}{ n l}=\sqrt{l+1} A_{n^{\prime} l^{\prime} n!} \delta_{l^{\prime} l+1}-\sqrt{l} B_{n^{\prime} l^{\prime} n!} \delta_{l^{\prime} l-1} \label{eq:13:2:22}
\end{equation}
The quantities \(A_{n^{\prime} l^{\prime} n!}\) and \(B_{n^{\prime} l^{\prime} n!}\) in Eq. \eqref{eq:13:2:22} may be represented as the integrals
\begin{align}
& A_{n^{\prime} l^{\prime} n l}=\int_{0}^{\infty} \Psi_{n^{\prime} l^{\prime}}^{*}(r)\left(\frac{\partial}{\partial r}-\frac{l}{r}\right) \Psi_{n l}(r) r^{2} d r \notag\\
& B_{n^{\prime} l^{\prime} n l}=\int_{0}^{\infty} \Psi_{n^{\prime} l^{\prime}}^{}(r)\left(\frac{\partial}{\partial r}+\frac{l+1}{r}\right) \Psi_{n l}(r) r^{2} d r \label{eq:13:2:23}
\end{align}
Here \(\Psi_{n l}(r)\) is the radial part of the corresponding wave function.\\
Reduced matrix elements of the operator \(\widehat{\nabla}_{1}(r, \vartheta, \varphi)\) in the (nlsJM)-representation may be obtained from Eq. \eqref{eq:13:2:5} by replacing \(\hat{\mathfrak{M}}_{k} \rightarrow \hat{\nabla}_{1}\).

The matrix elements \eqref{eq:13:2:21} vanish unless \(l^{\prime}=l+1\). In detailed form Eq. \eqref{eq:13:2:21} reads
\begin{align}
\braOket{ n^{\prime} l+1 m^{\prime}}{ \hat{\nabla}_{1 \pm 1}}{n l m} & =\sqrt{\frac{(l \pm m+1)(l \pm m+2)}{2(2 l+1)(2 l+3)}} A_{n^{\prime} l+1, n l} \delta_{m^{\prime} m \pm 1} \notag\\
\braOket{ n^{\prime} l+1 m^{\prime}}{ \hat{\nabla}_{10}}{n l m} & =\sqrt{\frac{(l-m+1)(l+m+1)}{(2 l+1)(2 l+3)}} A_{n^{\prime} l+1, n l} \delta_{m^{\prime} m} \\
\braOket{ n^{\prime} l-1 m^{\prime}}{ \hat{\nabla}_{1 \pm 1}}{n l m} & =-\sqrt{\frac{(l \mp m-1)(l \mp m)}{2(2 l+1)(2 l-1)}} B_{n^{\prime} l-1, n l} \delta_{m^{\prime} m \pm 1}  \label{eq:13:2:24}\\
\braOket{ n^{\prime} l-1 m^{\prime}}{ \hat{\nabla}_{10}}{n l m} & =\sqrt{\frac{(l-m)(l+m)}{(2 l+1)(2 l-1)} B_{n^{\prime} l-1, n l} \delta_{m^{\prime} m}}, \notag
\end{align}
All other matrix elements equal zero. The matrix elements of the cartesian components of \(\widehat{\nabla}_{1}(r, \vartheta, \varphi)\) are as\\
follows:
\begin{align}
\braOket{ n^{\prime} l^{\prime} m \pm 1}{ \hat{\nabla}_{x}}{n l m}= & \mp \frac{1}{2} \sqrt{\frac{(l \pm m+1)(l \pm m+2)}{(2 l+1)(2 l+3)}} A_{n^{\prime} l+1, n l} \delta_{l^{\prime} l+1}\notag \\
& \pm \frac{1}{2} \sqrt{\frac{(l \mp m-1)(l \mp m)}{(2 l+1)(2 l-1)}} B_{n^{\prime} l-1, n l} \delta_{l^{\prime} l-1}\notag \\
\braOket{ n^{\prime} l^{\prime} m \pm 1}{ \hat{\nabla}_{y}}{n l m}= & \frac{i}{2} \sqrt{\frac{(l \pm m+1)(l \pm m+2)}{(2 l+1)(2 l+3)}} A_{n^{\prime} l+1, n l} \delta_{l^{\prime} l+1} \notag\\
& -\frac{i}{2} \sqrt{\frac{(l \mp m-1)(l \mp m)}{(2 l+1)(2 l-1)}} B_{n^{\prime} l-1, n l} \delta_{l^{\prime} l-1}  \label{eq:13:2:25}\\
\braOket{ n^{\prime} l^{\prime} m}{ \hat{\nabla}_{z}}{n l m}= & \sqrt{\frac{(l-m+1)(l+m+1)}{(2 l+1)(2 l+3)}} A_{n^{\prime} l+1, n l} \delta_{l^{\prime} l+1} \\
& +\sqrt{\frac{(l-m)(l+m)}{(2 l+1)(2 l-1)} B_{n^{\prime} l-1, n l} \delta_{l^{\prime} l-1}}
. \notag
\end{align}
All other matrix elements are equal zero.\\
The matrix elements of the angular part of \(\widehat{\nabla}(r, \vartheta, \varphi)\) may be presented in a simpler form. According to Eq.~\eqref{eq:1:3:10},

\begin{equation}
\hat{\nabla}_{\Omega}=-i[\widehat{\vect{n}} \times \hat{\vect{L}}]
\end{equation}

This operator is independent of \(r\). The use of the Wigner-Eckart theorem yields the following expression for the matrix elements of the spherical components of \(\left(\hat{\nabla}_{\Omega}\right)_{1}\),
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left(\widehat{\nabla}_{\Omega}\right)_{1 \mu}}{l m}=\frac{\braOketred{ l^{\prime}}{\left(\widehat{\nabla}_{\Omega}\right)_{1}}{ l}}{\sqrt{2 l^{\prime}+1}} 
\clebsch{l}{m}{1}{\mu}{\prime^{\prime}}{m^{\prime}} \label{eq:13:2:26}
\end{equation}
where
\begin{equation}
\braOketred{ l^{\prime}}{\left(\hat{\nabla}_{\Omega}\right)_{1}}{ l}=-\left\{l \sqrt{l+1} \delta_{l^{\prime} l+1}+(l+1) \sqrt{l} \delta_{l^{\prime} l-1}\right\} \label{eq:13:2:27}
\end{equation}
Equation \eqref{eq:13:2:27} may be obtained from \eqref{eq:13:2:22} by the formal substitution
\begin{equation}
A_{n^{\prime} l^{\prime}, n l} \rightarrow-l, \quad B_{n^{\prime} l^{\prime}, n l} \rightarrow l+1 . \label{eq:13:2:28}
\end{equation}
Similarly, Eqs. \eqref{eq:13:2:24} and \eqref{eq:13:2:25} become valid for \(\left(\widehat{\nabla}_{\Omega}\right)_{1}\) after the same substitution \eqref{eq:13:2:28}.\\
Let us consider the matrix elements of products of two operators \(\left(\widehat{\nabla}_{\Omega}\right)_{1}\). In the case of the direct product we obtain
\begin{align}
& \braOket{ l^{\prime} m^{\prime}}{\left(\hat{\nabla}_{\Omega}\right)_{1 \mu}\left(\hat{\nabla}_{\Omega}\right)_{1 \nu}}{l m}=\frac{(-1)^{1-\mu}}{3} l(l+1) \delta_{l l^{\prime}} \delta_{m m^{\prime}} \delta_{\mu-\nu}+
\clebsch{1}{\mu}{1}{\nu}{2}{\kappa}\left\{\frac{l(l+1) \sqrt{(l+1)(l+2)}}{\sqrt{(2 l+3)(2 l+5)}} 
\clebsch{l}{m}{2}{\kappa}{l+2}{m+\kappa} \delta_{l^{\prime} l+2}\right. \notag\\
& \left.\quad+\frac{1}{2 l+1} \sqrt{\frac{l(l+1)}{6(2 l-1)(2 l+3)}}\left[4 l^{3}+6 l^{2}-4 l-3\right] 
\clebsch{l}{m}{2}{\kappa}{l}{m+\kappa} \delta_{l l^{\prime}}+\frac{l(l+1) \sqrt{l(l-1)}}{\sqrt{(2 l-1)(2 l-3)}} 
\clebsch{l}{m}{2}{\kappa}{l-2}{m+\kappa} \delta_{l^{\prime} l-2}\right\} \label{eq:13:2:29}
\end{align}
The matrix elements of the product of cartesian components \(\left(\widehat{\nabla}_{\Omega}\right)_{i}\left(\widehat{\nabla}_{\Omega}\right)_{k}(i, k=x, y, z)\) are related to the matrix elements \eqref{eq:13:2:29} by means of Eqs. \eqref{eq:13:2:18} where one should make the replacement \(\hat{n}_{i} \hat{n}_{k} \rightarrow\left(\hat{\nabla}_{\Omega}\right)_{i}\left(\hat{\nabla}_{\Omega}\right)_{k}(i, k=x, y, z)\). In the case of the irreducible tensor product of two operators \(\left(\widehat{\vec{\nabla}}_{\Omega}\right)_{1}\) one has
\begin{gather}
\braOket{ l^{\prime} m^{\prime}}{\left\{\left(\hat{\nabla}_{\Omega}\right)_{1} \otimes\left(\widehat{\nabla}_{\Omega}\right)_{1}\right\}_{k \kappa}}{l m} \notag\\
=(-1)^{k+l+l^{\prime}} \sqrt{\frac{2 k+l}{2 l^{\prime}+1}} 
\clebsch{l}{m}{k}{\kappa}{l^{\prime}}{m^{\prime}} \sum_{l^{\prime \prime}}\sixj{1}{1}{k}{l^{\prime}}{l}{l^{\prime \prime}}\braOketred{ l^{\prime}}{\left(\widehat{\nabla}_{\Omega}\right)_{1}}{ l^{\prime \prime}}\braOketred{ l^{\prime \prime}}{\left(\widehat{\nabla}_{\Omega}\right)_{1}}{ l} . \label{eq:13:2:30}
\end{gather}
In particular, Eq. \eqref{eq:13:2:30} yields the well known expression for the matrix elements of scalar product of two operators \(\left(\hat{\nabla}_{\Omega}\right)_{1}\)

\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left(\left(\hat{\nabla}_{\Omega}\right)_{1} \cdot\left(\hat{\nabla}_{\Omega}\right)_{1}\right)}{l m}=-\delta_{l^{\prime} l} \delta_{m^{\prime} m} l(l+1)
\end{equation}

Matrix elements of products of the operators \(\widehat{\vect{n}}_{1}\) and \(\left(\widehat{\boldsymbol{\nabla}}_{\Omega}\right)_{1}\) are of interest for application. For the matrix elements of the direct product of these operators one obtains
\begin{align}
& \braOket{ l^{\prime} m^{\prime}}{ \hat{n}_{1 \mu}\left(\hat{\nabla}_{\Omega}\right)_{1 \nu}}{l m}=-\delta_{l^{\prime}} \sqrt{\frac{l(l+1)}{2}}\left\{
\clebsch{1}{\mu}{1}{\nu}{1}{\kappa} 
\clebsch{l}{m}{1}{\kappa}{l}{m^{\prime}}+\sqrt{\frac{3}{(2 l-1)(2 l+3)}} 
\clebsch{1}{\mu}{1}{\nu}{2}{\kappa} 
\clebsch{l}{m}{2}{\kappa}{l}{m^{\prime}}\right\} \notag\\
& -\delta_{l^{\prime} l+2} l \sqrt{\frac{(l+1)(l+2)}{(2 l+3)(2 l+5)}} 
\clebsch{1}{\mu}{1}{\nu}{2}{\kappa} 
\clebsch{l}{m}{2}{\kappa}{l^{\prime}}{m^{\prime}}+\delta_{l^{\prime l}-2}(l+1) \sqrt{\frac{l(l-1)}{(2 l-1)(2 l-3)}} 
\clebsch{1}{\mu}{1}{\nu}{2}{\kappa} 
\clebsch{l}{m}{2}{\kappa}{l^{\prime}}{m^{\prime}} \label{eq:13:2:31}
\end{align}
The matrix element of the direct product \(\left(\widehat{\nabla}_{\Omega}\right)_{1 \mu} \widehat{n}_{1 \nu}\) differs from the foregoing one by the factor

\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{ \mathfrak{R}_{1 \nu 1 \mu}\left(\hat{n}_{1 \nu,}\left(\nabla_{\Omega}\right)_{1 \mu}\right)}{l m}
\end{equation}

where \(\mathfrak{R}_{1 \nu 1 \mu}\) is the commutator of the operator components in question (see Sec.~\ref{sec:3.1.7})
\begin{equation}
\mathfrak{R}_{1 \nu 1 \mu}\left(\hat{n}_{1 \nu},\left(\hat{\nabla}_{\Omega}\right)_{1 \mu}\right)=-\left(\hat{\nabla}_{\Omega}\right)_{1 \mu} \hat{n}_{1 \nu}=(-1)^{1+\nu} \frac{2}{3} \delta_{\nu-\mu}+2 \sqrt{\frac{2 \pi}{15}} 
\clebsch{1}{\mu}{1}{\nu}{2}{\mu+\nu} Y_{2 \mu+\nu}(\vartheta, \varphi) \label{eq:13:2:32}
\end{equation}
and
\begin{equation}
\braOket{l^{\prime} m^{\prime}}{\mathfrak{R}_{1 \mu 1 \nu}\left(\widehat{n}_{1 \nu},\left(\widehat{\nabla}_{\Omega}\right)_{1 \mu}\right)}{l m}=(-1)^{1+\nu} \frac{2}{3} \delta_{\nu-\mu} \delta_{l l^{\prime}} \delta_{m m^{\prime}}+\sqrt{\frac{2}{3} \frac{2 l+1}{2 l^{\prime}+1}} 
\clebsch{l}{0}{2}{0}{l^{\prime}}{0} 
\clebsch{1}{\mu}{1}{\nu}{2}{\mu+\nu} 
\clebsch{l}{m}{2}{\mu+\nu}{\prime^{\prime}}{m^{\prime}} \tag{13:2:32a}
\end{equation}
To determine the matrix elements of products of cartesian components of the operators \(\hat{\vect{n}}_{1}\) and \(\left(\hat{\nabla}_{\Omega}\right)_{1}\) one can transform these products to the products of spherical components (see Eq.~\eqref{eq:1:2:11}) and use Eq. \eqref{eq:13:2:31}.

For the irreducible tensor product of \(\hat{\vect{n}}_{1}(\vartheta, \varphi)\) and \(\left(\hat{\nabla}_{\Omega}\right)_{1}\) one obtains

\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left\{\hat{\mathrm{n}}_{1} \otimes\left(\hat{\nabla}_{\Omega}\right)_{1}\right\}_{k \kappa}}{l m}=(-1)^{l^{\prime}+l+k} 
\clebsch{l}{m}{k}{\kappa}{l^{\prime}}{m^{\prime}} \sqrt{\frac{2 k+1}{2 l^{\prime}+1}} \sum_{l^{\prime \prime}} \sqrt{2 l^{\prime \prime}+1}\sixj{1}{l}{l^{\prime \prime}}{l^\prime}{1}{k} \quad 
\clebsch{l^{\prime\prime}}{0}{1}{0}{l^{\prime}}{0} \braOketred{ l^{\prime \prime}}{\left(\hat{\nabla}_{0}\right)_{1}}{ l} .\label{eq:13:2:33}
\end{equation}

The reduced matrix element of \(\left(\widehat{\nabla}_{\Omega}\right)_{1}\) in \eqref{eq:13:2:33} is given by Eq. \eqref{eq:13:2:27}
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{\left\{\left(\widehat{\nabla}_{\Omega}\right)_{1} \otimes \widehat{\vect{n}}_{1}\right\}_{k c}}{l m}=(-1)^{k}\braOket{ l^{\prime} m^{\prime}}{\left\{\widehat{\vect{n}}_{1} \otimes\left(\nabla_{\Omega}\right)_{1}\right\}_{k c}}{l m}-(-1)^{k}\braOket{ l^{\prime} m^{\prime}}{ \mathfrak{R}_{k c}^{11}\left(\widehat{\vect{n}}_{1},\left(\widehat{\nabla}_{\Omega}\right)_{1}\right)}{l m} \label{eq:13:2:34}
\end{equation}
The second term in Eq. \eqref{eq:13:2:34} is the matrix element of the commutator of the irreducible tensor product in question.
\begin{equation}
\mathfrak{R}_{k \kappa}^{11}\left(\hat{\vect{n}}_{1},\left(\hat{\nabla}_{\Omega}\right)_{1}\right)=\frac{2}{\sqrt{3}} \delta_{k 0}+2 \sqrt{\frac{2 \pi}{15}} Y_{2 \kappa}(\vartheta, \varphi) \delta_{k 2} \label{eq:13:2:35}
\end{equation}
Using Eqs. \eqref{eq:13:2:104} and \eqref{eq:13:2:105}, we find
\begin{equation}
\braOket{ l^{\prime} m^{\prime}}{ \mathfrak{R}_{k \kappa}^{11}\left(\widehat{\vect{n}}_{1},\left(\widehat{\nabla}_{\Omega}\right)_{1}\right)}{l m}=\frac{2}{\sqrt{3}} \delta_{k 0} \delta_{l l} \delta_{m m^{\prime}}+\sqrt{\frac{2}{3} \frac{2 l+1}{2 l^{\prime}+1}} 
\clebsch{l}{0}{2}{0}{xxx^{\prime\prime}}{0}
\clebsch{l}{m}{2}{\kappa}{xxx^{\prime\prime}}{m^{\prime}} \delta_{k 2} \label{eq:13:2:36}
\end{equation}
\subsection{Matrix Elements of the Total Angular Momentum Operator \(\widehat{\vect{J}} \equiv \widehat{\vect{J}}_{1}\)}
The operator of the total angular momentum \(\widehat{\vect{J}}=\widehat{\vect{L}}+\widehat{\vect{S}}\) acts on position as well as spin variables. Making use of the Wigner-Eckart theorem, we get
\begin{equation}
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \hat{J}_{1 \mu}}{l s J M}=\frac{\braOketred{ l^{\prime} s^{\prime} J^{\prime}}{\widehat{\vect{J}}_{1}}{ l s J}}{\sqrt{2 J^{\prime}+1}} 
\clebsch{J}{M}{1}{\mu}{J^{\prime}}{M^{\prime}} \label{eq:13:2:37}
\end{equation}
where
\begin{equation}
\braOketred{ l^{\prime} s^{\prime} J^{\prime}}{\widehat{\vect{J}}_{1}}{ l s J}=\delta_{l l^{\prime}} \delta_{s s^{\prime}} \delta_{J J^{\prime}} \sqrt{J(J+1)(2 J+1)} \label{eq:13:2:38}
\end{equation}
For simplicity in equations \eqref{eq:13:2:41}-\eqref{eq:13:2:47} we shall omit the quantum numbers \(l\) and \(s\) in notation for matrix elements. Thus,
\begin{equation}
\braOket{ J^{\prime} M^{\prime}}{ \widehat{J}_{1 \mu}}{J M}=\frac{\braOketred{ J^{\prime}}{\widehat{\vect{J}}_{1}}{ J}}{\sqrt{2 J^{\prime}+1}} 
\clebsch{J}{M}{1}{\mu}{J^{\prime}}{M^{\prime}} \label{eq:13:2:39}
\end{equation}
where
\begin{equation}
\braOketred{ J^{\prime}}{\widehat{\vect{J}}_{1}}{ J}=\delta_{J J^{\prime}} \sqrt{J(J+1)(2 J+1)} \label{eq:13:2:40}
\end{equation}
In expanded form
\begin{align}
\braOket{ J M \pm 1}{ \widehat{J}_{1 \pm 1}}{J M} & =\mp \sqrt{\frac{(J \pm M+1)(J \mp M)}{2}}  \label{eq:13:2:41}\\
\braOket{ J M}{ \widehat{J}_{10}}{J M} & =M \notag
\end{align}
All other matrix elements are equal to zero. Let us present the non-vanishing matrix elements of the cartesian components of \(\hat{\vect{J}}\) :
\begin{align}
\braOket{ J M \pm 1}{ \widehat{J}_{x}}{J M} & =\frac{1}{2} \sqrt{(J \pm M+1)(J \mp M)}\notag \\
\braOket{ J M \pm 1}{ \widehat{J}_{y}}{J M} & =\mp \frac{i}{2} \sqrt{(J \pm M+1)(J \mp M)}  \label{eq:13:2:42}\\
\braOket{ J M}{ \widehat{J}_{z}}{J M} & =M. \notag
\end{align}
The matrix element of the operator \(\widehat{J}^{2}\) is as follows:
\begin{equation}
\braOket{ J^{\prime} M^{\prime}}{ \hat{\vect{J}}^{2}}{J M}=\delta_{J J^{\prime}} \delta_{M M^{\prime}} J(J+1) \label{eq:13:2:43}
\end{equation}
Let us consider matrix elements of products of the \(\widehat{\vect{J}}_{1}\) components. Matrix elements of direct product of spherical components may be represented in the form
\begin{equation}
\braOket{ J^{\prime} M^{\prime}}{ \hat{J}_{1 \mu} \hat{J}_{1 \nu}}{J M}=J(J+1) 
\clebsch{J}{M^{\prime}-\mu}{1}{\mu}{J}{M^{\prime}} 
\clebsch{J}{M}{1}{\nu}{J}{M^{\prime}-\mu} \delta_{J J^{\prime}} \label{eq:13:2:44}
\end{equation}
In component form
\begin{align}
\braOket{ J M \pm 2}{ \widehat{J}_{1 \pm 1} \widehat{J}_{1 \pm 1}}{J M} & =\frac{1}{2} \sqrt{(J \pm M+2)(J \pm M+1)(J \mp M-1)(J \mp M)} \notag\\
\braOket{ J M \pm 1}{ \widehat{J}_{1 \pm 1} \widehat{J}_{10}}{J M} & =\mp M \sqrt{\frac{(J \pm M+1)(J \mp M)}{2}} \notag\\
\braOket{ J M \pm 1}{ \widehat{J}_{10} \widehat{J}_{1 \pm 1}}{J M} & =\mp(M \pm 1) \sqrt{\frac{(J \pm M+1)(J \mp M)}{2}}  \label{eq:13:2:45}\\
\braOket{ J M}{ \widehat{J}_{1 \pm 1} \widehat{J}_{1 \mp 1}}{J M} & =-\frac{1}{2}(J \pm M)(J \mp M+1) \notag\\
\braOket{ J M}{ \widehat{J}_{10} \widehat{J}_{10}}{J M} & =M^{2}. \notag
\end{align}
The non-vanishing matrix elements of product of the cartesian components are as follows:
\begin{align}
\braOket{ J M \pm 2}{ \hat{J}_{x} \hat{J}_{x}}{J M} & =\frac{1}{4} \sqrt{(J \pm M+2)(J \pm M+1)(J \mp M-1)(J \mp M)} \notag\\
\braOket{ J M}{ \hat{J}_{x} \hat{J}_{x}}{J M} & =\frac{1}{2}\left[J(J+1)-M^{2}\right] \\
\braOket{ J M \pm 2}{ \hat{J}_{y} \hat{J}_{y}}{J M} & =-\frac{1}{4} \sqrt{(J \pm M+2)(J \pm M+1)(J \mp M-1)(J \mp M)} \notag\\
\braOket{ J M}{ \hat{J}_{y} \hat{J}_{y}}{J M} & =\frac{1}{2}\left[J(J+1)-M^{2}\right] \\
\braOket{ J M}{ \hat{J}_{z} \hat{J}_{z}}{J M} & =M^{2} \\
\braOket{ J M \pm 2}{ \hat{J}_{x} \hat{J}_{y}}{J M} & =\mp \frac{i}{4} \sqrt{(J \pm M+2)(J \pm M+1)(J \mp M-1)(J \mp M)} \notag\\
\braOket{ J M}{ \hat{J}_{x} \hat{J}_{y}}{J M} & =\frac{i}{2} M  \label{eq:13:2:46}\\
\braOket{ J M \pm 2}{ \hat{J}_{y} \hat{J}_{x}}{J M} & =\mp \frac{i}{4} \sqrt{(J \pm M+2)(J \pm M+1)(J \mp M-1)(J \mp M)} \notag\\
\braOket{ J M}{ \hat{J}_{y} \hat{J}_{x}}{J M} & =-\frac{i}{2} M, \notag\\
\braOket{ J M \pm 1}{ \hat{J}_{x} \hat{J}_{z}}{J M} & =\frac{M}{2} \sqrt{(J \pm M+1)(J \mp M)} \notag\\
\braOket{ J M \pm 1}{ \hat{J}_{z} \hat{J}_{x}}{J M} & =\frac{M \pm 1}{2} \sqrt{(J \pm M+1)(J \mp M)} \notag\\
\braOket{ J M \pm 1}{ \hat{J}_{y} \hat{J}_{z}}{J M} & =\mp \frac{i}{2} M \sqrt{(J \pm M+1)(J \mp M)} \notag\\
\braOket{ J M \pm 1}{ \hat{J}_{z} \hat{J}_{y}}{J M} & =\mp \frac{i}{2}(M \pm 1) \sqrt{(J \pm M+1)(J \mp M)}. \notag
\end{align}
In the case of the irreducible tensor product of two operators \(\widehat{\vect{J}}\) one obtains

\[
\braOket{ J^{\prime} M^{\prime}}{\left\{\widehat{\vect{J}}_{1} \otimes \widehat{\vect{J}}_{1}\right\}_{k c}}{J M}=\delta_{J J^{\prime}}(-1)^{2 J+k} J(J+1) \sqrt{(2 k+1)(2 J+1)}\sixj{1}{1}{k}{J}{J}{J}\label{eq:13:2:47} 
\clebsch{J}{M}{k}{c}{J}{M^{\prime}}
\]

In particular, from this equation one has two obvious relations

\begin{equation}
\begin{aligned}
\left.\braOket{ J^{\prime} M^{\prime}}{ \hat{\vect{J}} \cdot \hat{\vect{J}}\right)}{J M} & =\delta_{J J^{\prime}} \delta_{M M^{\prime}} J(J+1) \\
\braOket{ J^{\prime} M^{\prime}}{}{\hat{\vect{J}} \times \hat{\vect{J}}|_{\kappa}|J M} & =\delta_{J J^{\prime}} i \sqrt{J(J+1)} C_{J M M^{\prime}}^{J M^{\prime}}
\end{aligned}
\end{equation}

In applications, the expressions for matrix elements of products of the operator \(\widehat{\vect{J}}\) and the operators \(\widehat{\vect{n}}_{1}\) or \(\left(\widehat{\nabla}_{\Omega}\right)_{1}\) are of particular interest. Some relations for the matrix elements in questions will be considered below. The matrix elements of the direct tensor product of operators \(\widehat{\vect{n}}_{1}\) and \(\widehat{\vect{J}}_{1}\) are given by
\begin{gather}
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \widehat{n}_{1 \mu} \widehat{J}_{1 \nu}}{l s J M}=\delta_{e^{\prime}}(-1)^{\rho+J+l^{\prime}+1} \sqrt{J(J+1)(2 l+1)(2 J+1)} 
\clebsch{1}{0}{1}{0}{l^{\prime}}{0}\sixj{l}{s}{J}{J^{\prime}}{1}{l^{\prime}} 
\clebsch{J}{M+\nu}{1}{\mu}{J^{\prime}}{M^{\prime}} 
\clebsch{J}{M}{1}{\nu}{J}{M+\nu}  \label{eq:13:2:48}\\
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \widehat{J}_{1 \nu} \widehat{n}_{1 \mu}}{l s J M}=\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \widehat{n}_{1 \mu} \widehat{J}_{1 \nu}}{l s J M}-\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \mathfrak{R}_{1 \mu 1 \nu}\left(n_{1 \mu}, J_{1 \nu}\right)}{l s J M} \label{eq:13:2:49}
\end{gather}
The commutator of components of these operators is as follows:
\begin{equation}
\mathfrak{R}_{1 \mu 1 \nu}\left(\widehat{n}_{1 \mu}, \widehat{J}_{1 \nu}\right)=-\sqrt{\frac{8 \pi}{3}} 
\clebsch{1}{\mu}{1}{\nu}{1}{\mu+\nu} Y_{1 \mu+\nu}(\vartheta, \varphi) \label{eq:13:2:50}
\end{equation}
Using Eqs. \eqref{eq:13:2:104} and \eqref{eq:13:2:105}, one finds

\begin{equation}
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \mathfrak{R}_{1 \mu 1 \nu}\left(\hat{n}_{1 \mu}, \widehat{J}_{1 \nu}\right)}{l s J M}=(-1)^{J+l^{\prime}+{ }^{*}} \delta_{s s^{\prime}} \sqrt{2(2 l+1)(2 J+1)} 
\clebsch{l}{0}{1}{0}{l^{\prime}}{0}\sixj{l}{s}{J}{J^{\prime}}{1}{l^{\prime}}\label{eq:13:2:51} 
\clebsch{1}{\mu}{1}{\nu}{1}{\mu+\nu} 
\clebsch{J}{M}{1}{\mu+\nu}{J^{\prime}}{M^{\prime}}
\end{equation}

In the case of the irreducible tensor product of the operators \(\widehat{\vect{n}}_{1}\) and \(\widehat{\vect{J}}_{1}\) one obtains
\begin{align}
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{\left\{\widehat{\vect{n}}_{1} \otimes \widehat{\vect{J}}_{1}\right\}_{k \kappa}}{l s J M}= & (-1)^{s+l^{\prime}-J^{\prime}+k+1} \delta_{s s^{\prime}} \sqrt{(2 l+1)(2 k+1) J(J+1)} \notag\\
& \times(2 J+1) 
\clebsch{1}{0}{1}{0}{l^{\prime}}{0}\sixj{1}{1}{k}{J^{\prime}}{J}{J}\sixj{l}{s}{J}{J^{\prime}}{1}{l^{\prime}} 
\clebsch{J}{M}{k}{\kappa}{J^{\prime}}{M^{\prime}} \label{eq:13:2:52}
\end{align}
The commutator of the irreducible tensor product may be expressed as
\begin{equation}
\mathfrak{R}_{k \kappa}^{11}\left(\hat{\mathrm{n}}_{1}, \hat{\mathrm{~J}}_{1}\right)=-\sqrt{\frac{8 \pi}{3}} Y_{1 \kappa}(\vartheta, \varphi) \delta_{k 1} \label{eq:13:2:53}
\end{equation}
Taking Eqs. \eqref{eq:13:2:104} and \eqref{eq:13:2:105} into account, we get

\[
\braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{ \mathfrak{R}_{k \kappa}^{11}\left(\widehat{\vect{n}}_{1}, \widehat{\vect{J}}_{1}\right)}{l s J M}=(-1)^{J+l^{\prime}+s} \delta_{s s^{\prime}} \sqrt{2(2 J+1)(2 l+1)}\sixj{l}{s}{J}{J^{\prime}}{1}{l^{\prime}}\label{eq:13:2:54} 
\clebsch{l}{0}{1}{0}{l^{\prime}}{0} 
\clebsch{J}{M}{1}{\kappa}{J^{\prime}}{M^{\prime}} \delta_{k 1}

This operator acts only on spin variables. Using the Wigner-Eckart theorem, we get
\begin{gather}
\braOket{ s^{\prime} m_{s}^{\prime}}{ \widehat{S}_{1 \mu}}{s m_{s}}=\frac{\braOketred{ s^{\prime}}{\widehat{\vect{S}}_{1}}{ s}}{\sqrt{2 s^{\prime}+1}} 
\clebsch{s}{m,}{1}{\mu}{s^{\prime}}{m^{\prime}}  \label{eq:13:2:94}\\

\end{gather}
Equations \eqref{eq:13:2:39}-\eqref{eq:13:2:47} were obtained for the matrix elements of the operator \(\widehat{\vect{J}}_{1}\) and they remain valid for the spin operator \(\widehat{\vect{S}}_{1}\) after the replacement

\begin{equation}
J \rightarrow s, \quad M \rightarrow m_{s}, \quad \widehat{\vect{J}}_{1} \rightarrow \widehat{\vect{S}}_{1}
\end{equation}

The use of Eqs. \eqref{eq:13:2:6} and \eqref{eq:13:2:95} yields an expression for the reduced matrix element of the spin operator in the ( \(l s J\) )-representation

\[
\braOketred{ l^{\prime} s^{\prime} J^{\prime}}{\widehat{\mathrm{S}}_{1}}{ l s J}=(-1)^{l+s+J^{\prime}+1} \delta_{l l^{\prime}} \delta_{s s^{\prime}} \sqrt{s(s+1)(2 s+1)(2 J+1)\left(2 J^{\prime}+1\right)}\sixj{s}{l}{J}{J^{\prime}}{1}{s}\label{eq:13:2:96}
\]

The operators \(\widehat{\vect{n}}_{1},\left(\widehat{\nabla}_{\Omega}\right)_{1}, \widehat{\vect{L}}_{1}\) commute with the spin operator \(\widehat{\vect{S}}_{1}\). Matrix elements of the irreducible tensor products of these operators and the spin operator are given by
\begin{align}
& \braOket{ l^{\prime} s^{\prime} J^{\prime} M^{\prime}}{\left\{\widehat{\vect{n}}_{1} \otimes \widehat{\vect{S}}_{1}\right\}_{k \kappa}}{l s J M}=\delta_{s s^{\prime}} \sqrt{(2 k+1)(2 J+1)(2 l+1)(2 s+1) s(s+1)}
