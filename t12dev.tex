% ===== Chapter 12 diagram primitives =====
\tikzset{
  dl/.style={thick},               % diagram j-line: thick, no arrow
  dlu/.style={ultra thick},   % j-summed line: ultra-thick
  ddot/.style={inner sep=0pt},
}
% \dtheta{x}{y}{toplab}{midlab}{botlab}[midstyle] -- closed 2-arc "theta" with a
% horizontal middle line; centred at (x,y). Middle style optional (dl default).
\newcommand{\dtheta}[6][dl]{%
  \coordinate(tC)at(#2,#3);
  \coordinate(tL)at($(tC)+(-0.62,0)$){};
  \coordinate(tR)at($(tC)+(0.62,0)$){};
  \draw[dl](tL)to[bend left=60]node[midway,above,inner sep=1pt]{$#4$}(tR);
  \draw[dl](tL)to[bend right=60]node[midway,above,inner sep=1pt]{$#6$}(tR);
  \draw[#1](tL)--node[midway,above]{$#5$}(tR);}
% \dlens{Lname}{Rname}{toplab}{botlab} -- two arcs between existing coords L,R,
% with labels centred above/below.
\newcommand{\dlens}[4]{%
  \draw[dl](#1)to[bend left=60] node[midway,above,inner sep=1pt]{$#3$}(#2);
  \draw[dl](#1)to[bend right=60]node[midway,above,inner sep=1pt]{$#4$}(#2);}
% \dsixjsq[vdstyle]{x}{y}{tl}{tr}{bl}{br}{hd}{vd} -- 6j as a square/diamond with
% both diagonals; edge labels tl,tr (upper) bl,br (lower); hd=horizontal diagonal,
% vd=vertical diagonal (its style optional, pass dlu when summed).
\newcommand{\dsixjsq}[9][dl]{%
  \coordinate(qc)at(#2,#3);
  \coordinate(qt)at($(qc)+(90:0.9)$);\coordinate(qb)at($(qc)+(-90:0.9)$);
  \coordinate(ql)at($(qc)+(180:0.9)$);\coordinate(qr)at($(qc)+(0:0.9)$);
  \draw[dl](ql)--(qt)node[midway,above left,inner sep=1pt]{$#4$};
  \draw[dl](qt)--(qr)node[midway,above right,inner sep=1pt]{$#5$};
  \draw[dl](ql)--(qb)node[midway,below left,inner sep=1pt]{$#6$};
  \draw[dl](qb)--(qr)node[midway,below right,inner sep=1pt]{$#7$};
  \draw[dl](ql)--(qr)node[pos=0.28,below,inner sep=1pt]{$#8$};
  \draw[#1](qt)--(qb)node[pos=0.32,right,inner sep=1pt]{$#9$};}
% \dsixjtri[spkstyle]{x}{y}{eL}{eR}{eB}{sT}{sBL}{sBR} -- 6j as a triangle with a
% central node; outer edges eL (upper-left) eR (upper-right) eB (base); inner
% spokes sT (to apex) sBL (to bottom-left) sBR (to bottom-right).
\newcommand{\dsixjtri}[9][dl]{%
  \coordinate(ct)at(#2,#3);
  \coordinate(vT)at($(ct)+(90:1)$);
  \coordinate(vL)at($(ct)+(210:1)$);
  \coordinate(vR)at($(ct)+(330:1)$);
  \draw[dl](vL)--(vT)node[midway,above left,inner sep=1pt]{$#4$};
  \draw[dl](vT)--(vR)node[midway,above right,inner sep=1pt]{$#5$};
  \draw[dl](vL)--(vR)node[midway,below,inner sep=1pt]{$#6$};
  \draw[#1](ct)--(vT)node[pos=0.5,right,inner sep=1pt]{$#7$};
  \draw[dl](ct)--(vL)node[pos=0.5,above,inner sep=1pt]{$#8$};
  \draw[dl](ct)--(vR)node[pos=0.5,above,inner sep=1pt]{$#9$};}

% ---------- eq (3): lens = line + theta ----------
\[
\begin{tikzpicture}[baseline=0pt,font=\small]
  \coordinate(L)at(0,0){};\coordinate(R)at(1.5,0){};
  \draw[dl](-0.95,0)--(L)node[at start,above]{$a\alpha$};
  \draw[dl](R)--(2.45,0)node[at end,above]{$a'\alpha'$};
  \dlens{L}{R}{p}{q}
\end{tikzpicture}
\;=\;
\begin{tikzpicture}[baseline=0pt,font=\small]
  \draw[dl](0,0)--(1.9,0)node[pos=0.1,above]{$a\alpha$}node[pos=0.9,above]{$a'\alpha'$};
  \dtheta{3.1}{0}{p}{a}{q}
\end{tikzpicture}
\]

% \dsq{x}{y}{top}{bot}{left}{right} -- axis-aligned square centred at (x,y);
% defines corner coords sqTL,sqTR,sqBL,sqBR so the caller can add corner legs.
\newcommand{\dsq}[6]{%
  \coordinate(sqc)at(#1,#2);
  \coordinate(sqTL)at($(sqc)+(135:0.92)$);\coordinate(sqTR)at($(sqc)+(45:0.92)$);
  \coordinate(sqBL)at($(sqc)+(225:0.92)$);\coordinate(sqBR)at($(sqc)+(-45:0.92)$);
  \draw[dl](sqTL)--(sqTR)node[midway,above,inner sep=1pt]{$#3$};
  \draw[dl](sqBL)--(sqBR)node[midway,below,inner sep=1pt]{$#4$};
  \draw[dl](sqTL)--(sqBL)node[midway,left,inner sep=1pt]{$#5$};
  \draw[dl](sqTR)--(sqBR)node[midway,right,inner sep=1pt]{$#6$};}
