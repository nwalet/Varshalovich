% flat-top 9j-hexagon helper (edges from top, clockwise: top,UR,LR,bot,LL,UL);
% draws 3 full diameters, caller passes labels. dhexflat{x}{y}{top}{ur}{lr}{bot}{ll}{ul}
\newcommand{\dhexflat}[8]{%
  \foreach \i in {0,...,5}{\coordinate(v\i) at ($(#1,#2)+(60*\i:1)$);}
  \draw[dl](v2)--(v1)node[midway,above,inner sep=1pt]{$#3$};
  \draw[dl](v1)--(v0)node[midway,above right,inner sep=1pt]{$#4$};
  \draw[dl](v0)--(v5)node[midway,below right,inner sep=1pt]{$#5$};
  \draw[dl](v5)--(v4)node[midway,below,inner sep=1pt]{$#6$};
  \draw[dl](v4)--(v3)node[midway,below left,inner sep=1pt]{$#7$};
  \draw[dl](v3)--(v2)node[midway,above left,inner sep=1pt]{$#8$};}
