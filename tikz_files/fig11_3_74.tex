      \draw[jmlinem] ([yshift=(-2.7)*4mm]M.east) -- ([yshift=(-1.7)*4mm,xshift=8mm]M.east) node[midway,below] {$j_n$};

      \draw[jmlinem] ([yshift=(-0.7)*4mm,xshift=8mm]M.east) -- ([yshift=(-1.7)*4mm,xshift=8mm]M.east) node[midway,right] {$X_k$};
        \draw[jmlinem] ([yshift=(3-1+0.7)*4mm]M.east) -- ([yshift=(3-2+0.7)*4mm,xshift=8mm]M.east) node[midway,above] {$j_1$};
