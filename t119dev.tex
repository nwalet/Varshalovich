% Table 11.9, "Elimination of a J=0 line in a diagram".
% LHS: M-box with j (top) and j' (bottom) converging on a + node, from which
% an arrowed line labelled J=0 leaves to the right.
% RHS: M-box with a closed j-loop, times a triangle with +,+ vertices on the
% left, - vertex on the right, top edge J, bottom edge j'.
\[
\begin{tikzpicture}[baseline=0,inner sep=1.5pt,font=\scriptsize]
  \node[opbox,minimum width=0.8cm,minimum height=0.8cm](M) at (0,0){\normalsize$\mathfrak{M}$};
  \coordinate(ma)at($(M.south east)!0.75!(M.north east)$);
  \coordinate(mb)at($(M.south east)!0.25!(M.north east)$);
  \node[dotnode,label={above:\plusmark}](p)at($(M.east)+(0.8,0)$){};
  \draw[jmlinem](ma)--(p)node[pos=0.4,above]{$j$};
  \draw[jmlinem](mb)--(p)node[pos=0.4,below]{$j'$};
  \draw[jmline](p)--+(1.2,0)node[midway,above]{$J=0$};
\end{tikzpicture}
\;=\;
\begin{tikzpicture}[baseline=0,inner sep=1.5pt,font=\scriptsize]
  \node[opbox,minimum width=0.8cm,minimum height=0.8cm](M) at (0,0){\normalsize$\mathfrak{M}$};
  \coordinate(ma)at($(M.south east)!0.75!(M.north east)$);
  \coordinate(mb)at($(M.south east)!0.25!(M.north east)$);
  \draw[jmlinem](ma)to[out=0,in=0,looseness=2.5](mb);
  \node at($(M.east)+(0.55,0)$){$j$};
  % triangle
  \node[dotnode,label={above:\plusmark}](ta)at(1.35,0.45){};
  \node[dotnode,label={below:\plusmark}](tb)at(1.35,-0.45){};
  \node[dotnode,label={right:\minusmark}](tr)at(2.2,0){};
  \draw[jmlinem](ta)--(tb);
  \draw[jmlinem](ta)--(tr)node[midway,above right]{$J$};
  \draw[jmlinem](tr)--(tb)node[midway,below right]{$j'$};
\end{tikzpicture}
\]
