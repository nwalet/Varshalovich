% perimeter-arrow loop: \ploop{center-coord}{radius}{label}
\newcommand{\ploop}[3]{%
  \draw[thick,decoration={markings,mark=at position 0.6 with {\arrow{Stealth[length=1.6mm]}}},postaction={decorate}](#1)circle(#2);
  \node at(#1){$#3$};
  \node[dotnode,label={90:\plusmark}]at($(#1)+(0,#2)$){};}
% LHS coupled box M^(i): six legs -> two white triangles (J,J') + straight k -> apex
\newcommand{\redlhs}[5]{% #1=box label #2..#5 = leg labels j_a j_b j_c j_d
\begin{tikzpicture}[baseline=-6pt,inner sep=2pt,font=\scriptsize]
  \node[draw,minimum width=2.2cm,minimum height=0.85cm](B){\normalsize$#1$};
  \coordinate(TL)at([xshift=-8mm,yshift=8mm]B.north);
  \coordinate(TR)at([xshift=8mm,yshift=8mm]B.north);
  \node[dotnode,label={90:\plusmark}](ap)at([yshift=18mm]B.north){};
  \draw[jmlinem]([xshift=-13mm]B.north)--(TL)node[pos=0.4,left]{$#2$};
  \draw[jmlinem]([xshift=-5mm]B.north)--(TL)node[pos=0.4,right]{$#3$};
  \draw[jmlinem]([xshift=5mm]B.north)--(TR)node[pos=0.4,left]{$#4$};
  \draw[jmlinem]([xshift=13mm]B.north)--(TR)node[pos=0.4,right]{$#5$};
  \draw[jmlinem](TL)--(ap)node[midway,left]{$J$};
  \draw[jmlinem](B.north)--(ap)node[pos=0.45,right]{$k$};
  \draw[jmlinem](TR)--(ap)node[midway,right]{$J'$};
  \wtri at(TL){};\node at([xshift=2mm]TL){\tiny\plusmark};
  \wtri at(TR){};\node at([xshift=2mm]TR){\tiny\plusmark};
\end{tikzpicture}}
% RHS vertical stack for reduced m.e. : #1 box label, #2/#3 tribox roof (ja,ja'),
% #4/#5 inner-6j pair, #6/#7 spectator-loop labels (jb, jb')
\newcommand{\redrhs}[7]{%
\begin{tikzpicture}[baseline=-30pt,inner sep=2pt,font=\scriptsize]
  \node[draw,minimum width=1.5cm,minimum height=0.8cm](M){\normalsize$#1$};
  \coordinate(ap)at($(M.north)+(0,0.75)$);
  \node[dotnode,label={0:\plusmark}]at(ap){};
  \draw[jmlinem](M.north west)--(ap)node[midway,left]{$#2$};
  \draw[jmlinem](M.north)--(ap)node[pos=0.35,right]{$k$};
  \draw[jmlinem](M.north east)--(ap)node[midway,right]{$#3$};
  \coordinate(BA)at($(ap)+(0,0.4)$);
  \draw[jmlinem](ap)--(BA);
  \coordinate(TL)at($(BA)+(-0.85,1.2)$);\coordinate(TR)at($(BA)+(0.85,1.2)$);
  \coordinate(Cc)at($(BA)+(0,0.72)$);
  \draw[jmlinem](TL)--(TR)node[midway,above]{$k$};
  \draw[jmlinem](TL)--(BA)node[pos=0.5,left]{$#4$};
  \draw[jmlinem](TR)--(BA)node[pos=0.5,right]{$#5$};
  \draw[jmlinem](Cc)--(TL);\draw[jmlinem](Cc)--(TR);
  \draw[jmlinem](Cc)--(BA)node[pos=0.45,right]{$#6$};
  \node[dotnode,label={-90:\minusmark}]at(BA){};
  \node[dotnode,label={90:\minusmark}]at(TL){};\node[dotnode,label={90:\plusmark}]at(TR){};
  \draw[jmlinem](TL)--($(TL)+(-0.32,0)$);\draw[jmlinem](TR)--($(TR)+(0.32,0)$);
  \ploop{$(TL)+(-0.55,0)$}{0.22}{J}
  \ploop{$(TR)+(0.55,0)$}{0.22}{J'}
  % spectator loop below box
  \coordinate(sl)at($(M.south)+(0,-0.55)$);
  \draw[jmlinem](M.south)--($(sl)+(0,0.26)$);
  \draw[thick,decoration={markings,mark=at position 0.6 with {\arrow{Stealth[length=1.6mm]}}},postaction={decorate}](sl)circle(0.26);
  \node at($(sl)+(-0.42,0)$){$#6$};\node at($(sl)+(0.42,0)$){$#7$};
  \node[dotnode,label={90:\plusmark}]at($(sl)+(0,0.26)$){};
  \node[dotnode,label={-90:\plusmark}]at($(sl)+(0,-0.26)$){};
\end{tikzpicture}}
