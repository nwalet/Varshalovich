\newcommand{\tribox}[7]{%
\node[draw,minimum width=1.0cm,minimum height=0.9cm](B){\normalsize$#1$};
\coordinate(A)at($(B.north)+(0,0.75)$);
\draw[#5](B.north west)--(A)node[midway,left]{$#2$};
\draw[#6]($(B.north)$)--(A)node[pos=0.2,right]{$#3$};
\draw[#7](B.north east)--(A)node[midway,right]{$#4$};
\node[dotnode,label={90:\plusmark}]at(A){};}
